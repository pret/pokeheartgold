139BOOL VoltorbFlipTaskEngine_Run(VoltorbFlipTaskEngine *workflow) {
140 VoltorbFlipTasks *vfWorkflows = workflow->tasks;
141 VoltorbFlipTask *vfWorkflow = &(*vfWorkflows)[workflow->curWorkflow];
142 VoltorbFlipTaskFunc task;
143
144 switch (workflow->workflowState) {
145 case 0:
146 task = vfWorkflow->setup;
147 if (task == NULL) {
148 workflow->workflowState = 1;
149 } else {
150 if (task(workflow, workflow->work)) {
151 workflow->taskState = 0;
152 workflow->workflowState = 1;
153 }
154 break;
155 }
156 case 1:
157 task = vfWorkflow->begin;
158 if (task != NULL) {
159 task(workflow, workflow->work);
160 }
161 workflow->workflowState = 2;
162 case 2:
163 GF_ASSERT(vfWorkflow->main != NULL);
164 task = vfWorkflow->main;
165 if (task(workflow, workflow->work)) {
166 workflow->taskState = 0;
167 if (vfWorkflow->end != NULL) {
168 task = vfWorkflow->end;
169 task(workflow, workflow->work);
170 }
171 workflow->workflowState = 3;
172 }
173 break;
174 case 3:
175 task = vfWorkflow->tidy;
176 if (task == NULL) {
177 workflow->workflowState = 4;
178 } else {
179 if (task(workflow, workflow->work)) {
180 workflow->taskState = 0;
181 workflow->workflowState = 4;
182 }
183 break;
184 }
185 case 4:
186 GF_ASSERT(workflow->nextWorkflow != WORKFLOW_NONE);
187 workflow->curWorkflow = workflow->nextWorkflow;
188 workflow->nextWorkflow = WORKFLOW_NONE;
189
190 if (workflow->curWorkflow == WORKFLOW_TERMINATE) {
191 return TRUE;
192 }
193
194 workflow->workflowState = 0;
195 workflow->taskState = 0;
196 break;
197 default:
198 GF_ASSERT(FALSE);
199 }
200 return FALSE;
201}
