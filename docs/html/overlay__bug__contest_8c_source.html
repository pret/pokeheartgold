63 int i, j, cur_max, score, score2;
64 u8 temp;
65 u8 rand;
66 BugContestant *player;
67
68 // Judge the player's caught Pokemon
69 player = &bugContest->contestants[BUGCONTESTANT_PLAYER];
70 player->id = BUGCONTESTANT_PLAYER;
71 player->score = BugContest_JudgePlayerMon(bugContest, bugContest->mon);

74 // Init the rankings list

78
79 // Sort the rankings list by selection sort
80 for (i = 0; i < BUGCONTESTANT_COUNT - 1; i++) {
81 temp = bugContest->ranking[i];
82 cur_max = i;
83 score = bugContest->contestants[bugContest->ranking[i]].score;
84 for (j = i + 1; j < BUGCONTESTANT_COUNT; j++) {
85 score2 = bugContest->contestants[bugContest->ranking[j]].score;
86 if (score2 >= score) {
87 cur_max = j;
88 score = score2;
89 }
90 }
91 bugContest->ranking[i] = bugContest->ranking[cur_max];
92 bugContest->ranking[cur_max] = temp;
93 }
94
95 // Did the player win?
96 bugContest->placement = 3;
97 for (i = 0; i < 3; i++) {
98 if (bugContest->ranking[i] == BUGCONTESTANT_PLAYER) {
99 bugContest->placement = i;
100 break;
101 }
102 }
103
104 // Assign the player's prize item
105 switch (bugContest->placement) {
106 case 0: // First Place
107 if (bugContest->national_dex) {
108 rand = LCRandom() % 10;
109 } else {
110 rand = 0;
111 }
112 if (rand < 6) {
113 bugContest->prize = ITEM_SUN_STONE + rand;
114 } else {
115 bugContest->prize = ITEM_SHINY_STONE + rand - 6;
116 }
117 break;
118 case 1:
119 bugContest->prize = ITEM_EVERSTONE;
120 break;
121 case 2:
122 bugContest->prize = ITEM_SITRUS_BERRY;
123 break;
124 default:
125 bugContest->prize = ITEM_SHED_SHELL;

314u16 BugContest_JudgePlayerMon(BugContest *bugContest, Pokemon *mon) {
315 u16 score = 0;
316 int i;
317 u16 species;
318 BUGMON *bugmon = NULL;
319 u16 stat_total;
320
321 if (!bugContest->caught_poke) {
322 return score;
323 }
324 species = GetMonData(mon, MON_DATA_SPECIES, NULL);
325 for (i = 0; i < BUGMON_COUNT; i++) {
326 if (bugContest->encounters[i].species == species) {
327 bugmon = &bugContest->encounters[i];
328 score += bugContest->encounters[i].score;
329 break;
330 }
331 }
332 if (bugmon == NULL) {
333 GF_ASSERT(FALSE);
334 return 0;
335 }
336
337 score += GetMonData(mon, MON_DATA_LEVEL, NULL) * 100 / bugmon->lvlmax;
338 stat_total = 0;
339 for (i = 0; i < NUM_STATS; i++) {
340 stat_total += GetMonData(mon, MON_DATA_HP_IV + i, NULL);
341 }
342 score += stat_total * 100 / (31 * NUM_STATS);
343 score += GetMonData(mon, MON_DATA_HP, NULL) * 100 / GetMonData(mon, MON_DATA_MAX_HP, NULL);
344 return score;
