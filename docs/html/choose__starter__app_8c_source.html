1071static int getInput(struct ChooseStarterAppWork *work) {
1072 int ret = 0;
1073 // Ignore the keypad if continuing touch input
1074 if (System_GetTouchHeld() == 0) {
1075 if (gSystem.newKeys & PAD_BUTTON_A) {
1076 // Advance the starter selection.
1077 // It takes one nav input (A, left, or right) plus two A presses.
1078 if (work->state == SELECT_STATE_CONFIRM) {
1079 // Second A press, choice locked in
1080 ret = CHOOSE_STARTER_INPUT_CHOSE_STARTER;
1081 } else if (work->state == SELECT_STATE_INSPECT) {

1084 } else {
1085 // This counts as the "nav" input.

1088 }
1089 } else if (gSystem.newKeys & PAD_BUTTON_B) {
1090 // B button declines the selection
1091 if (work->state == SELECT_STATE_CONFIRM) {
1092 ret = CHOOSE_STARTER_INPUT_BACKED_OUT;
1093 }
1094 } else if (gSystem.newKeys & PAD_KEY_LEFT) {
1095 // Rotate the turntable clockwise
1096 if (work->state != SELECT_STATE_CONFIRM) {
1097 ret = CHOOSE_STARTER_INPUT_CYCLE_CLOCKWISE;
1098 }
1099 } else if (gSystem.newKeys & PAD_KEY_RIGHT) {
1100 // Rotate the turntable counterclockwise
1101 if (work->state != SELECT_STATE_CONFIRM) {
1102 ret = CHOOSE_STARTER_INPUT_CYCLE_COUNTERCLOCKWISE;
1103 }
1104 }
1105 } else if (gSystem.heldKeys == 0 && System_GetTouchNew() != 0) {
1106 VecFx32 near, far; // wherever you are
1107 int num;
1108 fx32 radius;
1109 NNS_G3dScrPosToWorldLine(gSystem.touchX, gSystem.touchY, &near, &far);
1110 if (work->state == SELECT_STATE_CONFIRM) {
1111 radius = 6 * FX32_ONE;
1112 } else {
1113 radius = 13 * FX32_ONE;
1114 }

1119 ret = CHOOSE_STARTER_INPUT_CHOSE_STARTER;

1121 ret = CHOOSE_STARTER_INPUT_BACKED_OUT;

1127 ret = getRotateDirection(num, work->curSelection, work->ballWobbleState);
