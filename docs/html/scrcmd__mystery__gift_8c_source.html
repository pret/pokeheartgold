132BOOL ScrCmd_MysteryGift(ScriptContext *ctx) {
133 switch (ScriptReadHalfword(ctx)) {
134 case SCR_MG_BEGIN:
135 SaveMGDataPtr_Begin(ctx->fieldSystem->saveData, HEAP_ID_FIELD3);
136 break;
137 case SCR_MG_END:
138 SaveMGDataPtr_End(ctx->fieldSystem->saveData, FALSE);
139 break;
140 case SCR_MG_END2:
141 SaveMGDataPtr_End(ctx->fieldSystem->saveData, TRUE);
142 break;
143 case SCR_MG_HAS_GIFT: {
144 u16 *ptr = ScriptGetVarPointer(ctx);
145 if (FieldSystem_GetTagOfNextMG(ctx->fieldSystem) != MG_TAG_INVALID) {
146 *ptr = TRUE;
147 } else {
148 *ptr = FALSE;
149 }
150 break;
151 }
152 case SCR_MG_GET_TYPE: {
153 u16 *ptr = ScriptGetVarPointer(ctx);
154 *ptr = FieldSystem_GetTagOfNextMG(ctx->fieldSystem);
155 break;
156 }
157 case SCR_MG_CAN_RECEIVE: {
158 u16 *ptr = ScriptGetVarPointer(ctx);
159 const struct ScriptMysteryGiftFuncs *pFunc = &sScriptMysteryGiftActionTable[FieldSystem_GetTagOfNextMG(ctx->fieldSystem) - 1];
160 *ptr = pFunc->check(ctx->fieldSystem, FieldSystem_GetDataOfNextMG(ctx->fieldSystem));
161 break;
162 }
163 case SCR_MG_RECEIVE: {
164 const struct ScriptMysteryGiftFuncs *pFunc = &sScriptMysteryGiftActionTable[FieldSystem_GetTagOfNextMG(ctx->fieldSystem) - 1];
165 pFunc->give(ctx->fieldSystem, FieldSystem_GetDataOfNextMG(ctx->fieldSystem));
166 FieldSystem_SetQueuedMGReceived(ctx->fieldSystem);
167 break;
168 }
169 case SCR_MG_MESSAGE_RECEIVED: {
170 struct GetMysteryGiftGmmState gmmState;
171 const struct ScriptMysteryGiftFuncs *pFunc = &sScriptMysteryGiftActionTable[FieldSystem_GetTagOfNextMG(ctx->fieldSystem) - 1];
172 MessageFormat **pMsgFormat = FieldSysGetAttrAddr(ctx->fieldSystem, SCRIPTENV_MESSAGE_FORMAT);
173 u16 *pMsgBank = ScriptGetVarPointer(ctx);
174 u16 *pMsgNum = ScriptGetVarPointer(ctx);
175 FieldSystem_InitGetMysteryGiftGmmState(&gmmState, ctx->fieldSystem, *pMsgFormat, FieldSystem_GetDataOfNextMG(ctx->fieldSystem));
176 pFunc->messageSuccess(&gmmState, pMsgBank, pMsgNum);
177 break;
178 }
179 case SCR_MG_MESSAGE_FAILED: {
180 struct GetMysteryGiftGmmState gmmState;
181 const struct ScriptMysteryGiftFuncs *pFunc = &sScriptMysteryGiftActionTable[FieldSystem_GetTagOfNextMG(ctx->fieldSystem) - 1];
182 MessageFormat **pMsgFormat = FieldSysGetAttrAddr(ctx->fieldSystem, SCRIPTENV_MESSAGE_FORMAT);
183 u16 *pMsgBank = ScriptGetVarPointer(ctx);
184 u16 *pMsgNum = ScriptGetVarPointer(ctx);
185 FieldSystem_InitGetMysteryGiftGmmState(&gmmState, ctx->fieldSystem, *pMsgFormat, FieldSystem_GetDataOfNextMG(ctx->fieldSystem));
186 pFunc->messageFailure(&gmmState, pMsgBank, pMsgNum);
187 break;
188 }
189 }
190 return FALSE;
191}
