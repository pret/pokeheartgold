150static u16 GetMoveTutorLearnsetIndex(u16 species, u8 form) {
151 u16 index;
152 // subtract 2 because SPECIES_EGG and SPECIES_BAD_EGG are missing
153 switch (species) {
154 case SPECIES_DEOXYS:
155 if (form == DEOXYS_ATTACK) {
156 index = SPECIES_DEOXYS_ATK - 2;
157 } else if (form == DEOXYS_DEFENSE) {
158 index = SPECIES_DEOXYS_DEF - 2;
159 } else if (form == DEOXYS_SPEED) {
160 index = SPECIES_DEOXYS_SPD - 2;
161 }
162 break;
163 case SPECIES_WORMADAM:
164 if (form == WORMADAM_SANDY) {
165 index = SPECIES_WORMADAM_SANDY - 2;
166 } else if (form == WORMADAM_TRASH) {
167 index = SPECIES_WORMADAM_TRASH - 2;
168 }
169 break;
170 case SPECIES_GIRATINA:
171 if (form == GIRATINA_ORIGIN) {
172 index = SPECIES_GIRATINA_ORIGIN - 2;
173 }
174 break;
175 case SPECIES_SHAYMIN:
176 if (form == SHAYMIN_SKY) {
177 index = SPECIES_SHAYMIN_SKY - 2;
178 }
179 break;
180 case SPECIES_ROTOM:
181 if (form == ROTOM_HEAT) {
182 index = SPECIES_ROTOM_HEAT - 2;
183 } else if (form == ROTOM_WASH) {
184 index = SPECIES_ROTOM_WASH - 2;

186 index = SPECIES_ROTOM_FROST - 2;

189 } else if (form == ROTOM_MOW) {
190 index = SPECIES_ROTOM_MOW - 2;
191 }
192 break;
193 }

197static u16 GetLearnableTutorMoves(Pokemon *mon, u32 moveTutorNpc, u8 dest[]) {
198 s32 i, j;
199 u16 currentMoves[MAX_MON_MOVES];
200 for (i = 0; i < MAX_MON_MOVES; i++) {
201 currentMoves[i] = GetMonData(mon, MON_DATA_MOVE1 + i, NULL);
202 }
203 u32 species = GetMonData(mon, MON_DATA_SPECIES, NULL);
204 u32 form = GetMonData(mon, MON_DATA_FORM, NULL);
205 MoveTutorLearnset *learnset = GetMoveTutorLearnset(HEAP_ID_FIELD2, GetMoveTutorLearnsetIndex(species, form));
206 u16 numLearnableMoves = 0;
207 for (j = 0; j < NELEMS(sTutorMoves); j++) {
208 // this is equivalent to treating `learnset` as a bitfield of 64 bits
209 // and then checking whether bit j is set
210 BOOL canLearnMove = (learnset->moves[j / 8] >> (j % 8)) & 1;
211 if (canLearnMove && moveTutorNpc == sTutorMoves[j].tutorNpc) {
212 for (i = 0; i < MAX_MON_MOVES; i++) {
213 if (currentMoves[i] == sTutorMoves[j].move) {
214 break;
215 }
216 }
217 if (i < MAX_MON_MOVES) { // move already currently learned by pokemon
218 continue;
219 }
220 if (dest != NULL) {
221 dest[numLearnableMoves] = j;
222 }
223 numLearnableMoves++;
224 }
225 }
226 Heap_Free(learnset);
227 return numLearnableMoves;
228}

231 FSFile file;
232 FS_InitFile(&file);

234 GF_ASSERT(FALSE);

244 FS_ReadFile(&file, learnset, sizeof(MoveTutorLearnset));
245 FS_CloseFile(&file);
