43BOOL IntroMovie_Scene1(IntroMovieOverlayData *data, void *pVoid) {
44 IntroMovieScene1Data *sceneData = (IntroMovieScene1Data *)pVoid;
45
46 if (IntroMovie_GetIntroSkippedFlag(data)) {
47 sceneData->state = INTRO_MOVIE_SCENE_EXIT;
48 }
49
50 switch (sceneData->state) {
51 case INTRO_MOVIE_SCENE_INIT:
52 IntroMovie_Scene1_Init(data, sceneData);
53 ++sceneData->state;
54 break;
55 case INTRO_MOVIE_SCENE_RUN:
56 if (IntroMovie_Scene1_Main(data, sceneData, IntroMovie_GetTotalFrameCount(data))) {
57 ++sceneData->state;
58 }
59 break;
60 case INTRO_MOVIE_SCENE_EXIT:
61 IntroMovie_Scene1_Exit(data, sceneData);
62 return TRUE;
63 }
64
65 return FALSE;
66}

91static BOOL IntroMovie_Scene1_Main(IntroMovieOverlayData *data, IntroMovieScene1Data *sceneData, int totalFrames) {
92 BgConfig *bgConfig = IntroMovie_GetBgConfig(data);
93 IntroMovieBgLinearAnims *bgAnimCnt = IntroMovie_GetBgLinearAnimsController(data);
94 u8 timer = IntroMovie_GetSceneStepTimer(data);
95 switch (IntroMovie_GetSceneStep(data)) {
96 case INTRO_SCENE1_APPEAR_COPYRIGHT: // Copyright appear
97 BgSetPosTextAndCommit(bgConfig, GF_BG_LYR_MAIN_0, BG_POS_OP_SET_Y, 128);
98 GfGfx_EngineATogglePlanes(GX_PLANEMASK_BG0, GF_PLANE_TOGGLE_ON);
99 GfGfx_EngineBTogglePlanes(GX_PLANEMASK_BG0, GF_PLANE_TOGGLE_ON);
100 IntroMovie_AdvanceSceneStep(data);
101 break;
102 case INTRO_SCENE1_WAIT_COPYRIGHT: // Wait 30 frames then start fade out
103 if (timer >= 30) {
104 IntroMovie_StartBlendFadeEffect(&bgAnimCnt->blend[0], 1, 0x20, 60, 1, 0);
105 IntroMovie_StartBlendFadeEffect(&bgAnimCnt->blend[1], 1, 0x20, 60, 1, 1);
106 IntroMovie_AdvanceSceneStep(data);
107 }
108 break;
109 case INTRO_SCENE1_WAIT_FADEOUT_COPYRIGHT: // Wait for fade out (60 frames)
110 if (bgAnimCnt->blend[0].finished && bgAnimCnt->blend[1].finished) {
111 IntroMovie_AdvanceSceneStep(data);
112 }
113 break;
114 case INTRO_SCENE1_WAIT_APPEAR_GAMEFREAK: // Delay 20 frames then appear GameFreak logo
115 if (timer >= 20) {
116 *sceneData->skipAllowedPtr = TRUE;
117 GfGfx_EngineBTogglePlanes(GX_PLANEMASK_BG0, GF_PLANE_TOGGLE_OFF);
118 GfGfx_EngineBTogglePlanes(GX_PLANEMASK_BG1, GF_PLANE_TOGGLE_ON);
119 BgSetPosTextAndCommit(bgConfig, GF_BG_LYR_MAIN_0, BG_POS_OP_SET_Y, 0);
120 G2_SetBlendAlpha(1, 0x20, 0x1F, 0);
121 IntroMovie_AdvanceSceneStep(data);
122 }
123 break;
124 case INTRO_SCENE1_WAIT_GAMEFREAK: // Delay 110 frames
125 if (timer >= 110) {
126 IntroMovie_AdvanceSceneStep(data);
127 }
128 break;
129 case INTRO_SCENE1_APPEAR_BG_IMAGE: // Appear sunrise or sunset background
130 GfGfx_EngineATogglePlanes((GXPlaneMask)(GX_PLANEMASK_BG1 | GX_PLANEMASK_BG2 | GX_PLANEMASK_BG3), GF_PLANE_TOGGLE_ON);
131 GfGfx_EngineBTogglePlanes(GX_PLANEMASK_BG3, GF_PLANE_TOGGLE_ON);
132 Sprite_SetDrawFlag(sceneData->sunSprite, TRUE);
133 IntroMovie_AdvanceSceneStep(data);
134 break;
135 case INTRO_SCENE1_WAIT_START_BG_SCROLL: // Start scrolling background
136 if (timer >= 1) {
137 IntroMovie_StartBlendFadeEffect(&bgAnimCnt->blend[0], 1, 0x1E, 80, 1, 1);
138 IntroMovie_StartBlendFadeEffect(&bgAnimCnt->blend[1], 2, 0x1E, 80, 1, 0);
139 IntroMovie_StartBgScroll_VBlank(bgConfig, bgAnimCnt->scroll, GF_BG_LYR_MAIN_1, 0, -0x20, 0xF0);
140 IntroMovie_StartBgScroll_VBlank(bgConfig, bgAnimCnt->scroll, GF_BG_LYR_MAIN_2, 0, -0x10, 0xF0);
141 IntroMovie_StartSpriteAnimAndMakeVisible(sceneData->sunSprite, TRUE);
142 IntroMovie_AdvanceSceneStep(data);
143 }
144 break;
145 case INTRO_SCENE1_WAIT_BG_SCROLL: // Wait blend and scroll effects (240 frames)
146 if (bgAnimCnt->blend[0].finished && bgAnimCnt->blend[1].finished && IntroMovie_WaitBgScrollAnim(bgAnimCnt->scroll, GF_BG_LYR_MAIN_1) && IntroMovie_WaitBgScrollAnim(bgAnimCnt->scroll, GF_BG_LYR_MAIN_2)) {
147 IntroMovie_AdvanceSceneStep(data);
148 }
149 break;
150 case INTRO_SCENE1_WAIT_APPEAR_BIRD: // Delay 128 frames then start box legendary anim
151 if (timer >= 128) {
152 IntroMovie_StartSpriteAnimAndMakeVisible(sceneData->birdSprite, TRUE);
153 IntroMovie_AdvanceSceneStep(data);
154 }
155 break;
156 case INTRO_SCENE1_DELAY90_START_FADEOUT: // Run anim 90 frames then start fade to white
157 if (timer >= 90) {
158 BeginNormalPaletteFade(0, 0, 0, RGB_WHITE, 65, 1, HEAP_ID_INTRO_MOVIE);
159 IntroMovie_AdvanceSceneStep(data);
160 }
161 break;
162 case INTRO_SCENE1_WAIT_FADEOUT: // Wait fade to white (65 frames)
163 if (IsPaletteFadeFinished()) {
164 return TRUE;
165 }
166 }
167
168 return FALSE;
169}
