568s32 TryModEV(s32 ev, s32 evSum, s32 by) {
569 if (ev == 0 && by < 0) {
570 return -1;
571 }
572
573 if (ev >= MAX_EV_VITAMINS && by > 0) {
574 return -1;
575 }
576
577 if (ev + evSum >= MAX_EV_SUM && by > 0) {
578 return -1;
579 }
580
581 ev += by;
582 if (ev > MAX_EV_VITAMINS) {
583 ev = MAX_EV_VITAMINS;
584 } else if (ev < 0) {
585 ev = 0;
586 }
587
588 if (ev + evSum > MAX_EV_SUM) {
589 ev = MAX_EV_SUM - evSum;
590 }
591
592 return ev;
