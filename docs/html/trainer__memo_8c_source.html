302BOOL MonMetadataMatchesEvent(u8 eventNo, Pokemon *mon, BOOL isMine) {
303 u8 metCondition = MonMetCondition(mon, isMine);
304 u8 version = GetMonData(mon, MON_DATA_MET_GAME, NULL);
305
306 if ((metCondition == MET_CONDITION_FATEFUL_ENCOUNTER || metCondition == MET_CONDITION_FATEFUL_ENCOUNTER_TRADED) && (eventNo != EVENT_ARCEUS_HALL_OF_ORIGIN)) {
307 return TRUE;
308 }
309
310 switch (eventNo) {
311 case EVENT_SPIKY_EARED_PICHU:
312 break;
313 case EVENT_ARCEUS_HALL_OF_ORIGIN:
314 if (metCondition == 1) {
315 switch (version) {
316 case VERSION_DIAMOND:
317 case VERSION_PEARL:
318 case VERSION_PLATINUM:
319 if (GetMonData(mon, MON_DATA_MET_LOCATION, NULL) == sub_02017FE4(MAPSECTYPE_NORMAL, MAPSEC_HALL_OF_ORIGIN)) {
320 return TRUE;
321 }
322 }
323 }
324 break;
325 case EVENT_ARCEUS_MOVIE_GIFT:
326 case EVENT_CELEBI:
327 break;
328 }
