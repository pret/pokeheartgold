38static u16 PhoneScriptGeneric_GetScriptIdInternal(PokegearPhoneCallContext *ctx, const PhoneScriptGenericHeader *hdrs, int count) {
39 // at runtime, param:count is always 8
40 int i;
41 BOOL result;
42 PokegearPhoneCallState *state = &ctx->state;
43 const PhoneScriptGenericHeader *curHdr = hdrs;
