271void CreateNPCTrainerParty(BattleSetup *enemies, int partyIndex, enum HeapID heapID) {
272 TRPOKE *data; // sp74
273 int i;
274 int j;
275 u32 pidGender; // sp7C
276 u32 personality;
277 u32 seedBak;
278 u8 iv;
279 Pokemon *mon;
280
281 // We abuse the global RNG for personality value generation,
282 // so back up the overworld state here.
283 seedBak = GetLCRNGSeed();
284 Party_InitWithMaxSize(enemies->party[partyIndex], PARTY_SIZE);
285 data = (TRPOKE *)Heap_Alloc(heapID, sizeof(TRPOKE) * PARTY_SIZE);
286 mon = AllocMonZeroed(heapID);
287 TrainerData_ReadTrPoke(enemies->trainerId[partyIndex], data);
288
289 // If a Pokemon's gender ratio is 50/50, the generated Pokemon will be the same
290 // gender as its trainer. Otherwise, it will assume the more abundant gender
291 // according to its species gender ratio. In double battles, the behavior is
292 // identical to that of a solitary male opponent.
293 if (TrainerClass_GetGenderOrTrainerCount(enemies->trainer[partyIndex].data.trainerClass) == TRAINER_FEMALE) {
294 pidGender = 0x78;
295 } else {
296 pidGender = 0x88;
297 }
298
299 // The trainer types can be more efficiently and expandibly treated as a flag
300 // array, with bit 0 being custom moveset and bit 1 being held item.
301 // Game Freak didn't do it that way, instead using a switch statement and a lot
302 // of code duplication. This has been the case since the 2nd generation games.
303 switch (enemies->trainer[partyIndex].data.trainerType) {
304 case TRTYPE_MON: {
305 TRPOKE_NOITEM_DFLTMOVES *monSpecies;
306 u16 species;
307 u8 form;
308 monSpecies = &data->species;
309 for (i = 0; i < enemies->trainer[partyIndex].data.npoke; i++) {
310 // Starting in Platinum, the Pokemon's form was encoded
311 // in the upper 6 bits of the species.
312 species = monSpecies[i].species & 0x3FF;
313 form = (monSpecies[i].species & 0xFC00) >> 10;
314 // Starting in HGSS, additional checks are performed to
315 // rand each Pokemon's personality.
316 TrMon_OverridePidGender(species, form, monSpecies[i].genderAbilityOverride, &pidGender);
317 // Generate personality by seeding with a value based on the difficulty,
318 // level, species, and opponent ID. Roll the RNG N times, where N is
319 // the index of its trainer class. Finally, left shift the 16-bit
320 // pseudorandom value and add the gender selector.
321 // This guarantees that NPC trainer' Pokemon are generated in a
322 // consistent manner between attempts.
323 // This procedure results in only a 24-bit peersonality value.
324 personality = monSpecies[i].difficulty + monSpecies[i].level + species + enemies->trainerId[partyIndex];
325 SetLCRNGSeed(personality);
326 for (j = 0; j < enemies->trainer[partyIndex].data.trainerClass; j++) {
327 personality = LCRandom();
328 }
329 personality = (personality << 8) + pidGender;
330
331 // Difficulty is a number between 0 and 250 which directly corresponds
332 // to the (uniform) IV spread of the generated Pokemon.
333 iv = (u8)((monSpecies[i].difficulty * 31) / 255);
334 CreateMon(mon, species, monSpecies[i].level, iv, TRUE, (s32)personality, OT_ID_RANDOM_NO_SHINY, 0);
335
336 // If you were treating the trainer type as a bitfield, you'd put the
337 // checks for held item and moves here. You'd also treat the trpoke
338 // data as a flat u16 array rather than an array of fixed-width structs.
339
340 // Starting in Platinum, enemy trainer could have
341 // personalized ball capsules.
342 SetTrMonCapsule(monSpecies[i].capsule, mon, heapID);
343 SetMonData(mon, MON_DATA_FORM, &form);
344 // Starting in HGSS, an AI Pokemon with Frustration
345 // will have minimum friendship.
346 TrMon_FrustrationCheckAndSetFriendship(mon);
347 Party_AddMon(enemies->party[partyIndex], mon);
348 }
349 break;
350 }
351 case TRTYPE_MON_MOVES: {
352 TRPOKE_NOITEM_CUSTMOVES *monSpeciesMoves;
353 u16 species;
354 u8 form;
355 monSpeciesMoves = &data->species_moves;
356 for (i = 0; i < enemies->trainer[partyIndex].data.npoke; i++) {
357 species = monSpeciesMoves[i].species & 0x3FF;
358 form = (monSpeciesMoves[i].species & 0xFC00) >> 10;
359 TrMon_OverridePidGender(species, form, monSpeciesMoves[i].genderAbilityOverride, &pidGender);
360 personality = monSpeciesMoves[i].difficulty + monSpeciesMoves[i].level + species + enemies->trainerId[partyIndex];
361 SetLCRNGSeed(personality);
362 for (j = 0; j < enemies->trainer[partyIndex].data.trainerClass; j++) {
363 personality = LCRandom();
364 }
365 personality = (personality << 8) + pidGender;
366 iv = (u8)((monSpeciesMoves[i].difficulty * 31) / 255);
367 CreateMon(mon, species, monSpeciesMoves[i].level, iv, TRUE, (s32)personality, OT_ID_RANDOM_NO_SHINY, 0);
368 for (j = 0; j < MAX_MON_MOVES; j++) {
369 MonSetMoveInSlot(mon, monSpeciesMoves[i].moves[j], (u8)j);
370 }
371 SetTrMonCapsule(monSpeciesMoves[i].capsule, mon, heapID);
372 SetMonData(mon, MON_DATA_FORM, &form);
373 TrMon_FrustrationCheckAndSetFriendship(mon);
374 Party_AddMon(enemies->party[partyIndex], mon);
375 }
376 break;
377 }
378 case TRTYPE_MON_ITEM: {
379 TRPOKE_ITEM_DFLTMOVES *monSpeciesItem;
380 u16 species;
381 u8 form;
382 monSpeciesItem = &data->species_item;
383 for (i = 0; i < enemies->trainer[partyIndex].data.npoke; i++) {
384 species = monSpeciesItem[i].species & 0x3FF;
385 form = (monSpeciesItem[i].species & 0xFC00) >> 10;
386 TrMon_OverridePidGender(species, form, monSpeciesItem[i].genderAbilityOverride, &pidGender);
387 personality = monSpeciesItem[i].difficulty + monSpeciesItem[i].level + species + enemies->trainerId[partyIndex];
388 SetLCRNGSeed(personality);
389 for (j = 0; j < enemies->trainer[partyIndex].data.trainerClass; j++) {
390 personality = LCRandom();
391 }
392 personality = (personality << 8) + pidGender;
393 iv = (u8)((monSpeciesItem[i].difficulty * 31) / 255);

395 SetMonData(mon, MON_DATA_HELD_ITEM, &monSpeciesItem[i].item);
396 SetTrMonCapsule(monSpeciesItem[i].capsule, mon, heapID);
397 SetMonData(mon, MON_DATA_FORM, &form);
398 TrMon_FrustrationCheckAndSetFriendship(mon);

402 }
403 case TRTYPE_MON_ITEM_MOVES: {
404 TRPOKE_ITEM_CUSTMOVES *monSpeciesItemMoves;
405 u16 species;
406 u8 form;
407 monSpeciesItemMoves = &data->species_item_moves;
408 for (i = 0; i < enemies->trainer[partyIndex].data.npoke; i++) {
409 species = monSpeciesItemMoves[i].species & 0x3FF;
410 form = (monSpeciesItemMoves[i].species & 0xFC00) >> 10;
411 TrMon_OverridePidGender(species, form, monSpeciesItemMoves[i].genderAbilityOverride, &pidGender);
412 personality = monSpeciesItemMoves[i].difficulty + monSpeciesItemMoves[i].level + species + enemies->trainerId[partyIndex];
413 SetLCRNGSeed(personality);
414 for (j = 0; j < enemies->trainer[partyIndex].data.trainerClass; j++) {
415 personality = LCRandom();
416 }
417 personality = (personality << 8) + pidGender;
418 iv = (u8)((monSpeciesItemMoves[i].difficulty * 31) / 255);

437void TrMon_OverridePidGender(int species, int form, int overrideParam, u32 *pid) {
438 int genderOverride = overrideParam & 0xF;
439 int abilityOverride = (overrideParam & 0xF0) >> 4;
440 if (overrideParam != 0) {
441 if (genderOverride != 0) {
442 *pid = GetMonBaseStat_HandleAlternateForm(species, form, BASE_GENDER_RATIO);
443 if (genderOverride == 1) {
444 *pid += 2;
445 } else {
446 *pid -= 2;
447 }
448 }
449 if (abilityOverride == 1) {
450 *pid &= ~1;
451 } else if (abilityOverride == 2) {
452 *pid |= 1;
453 }
454 }
455}
