53int PokegearApp_HandleTouchInput_SwitchApps(PokegearAppData *pokegearApp) {
54 // Don't switch if didn't tap an app button
55 int newApp = TouchscreenHitbox_FindRectAtTouchNew(sTouchscreenButtonHitboxes);
56 if (newApp == TOUCH_MENU_NO_INPUT) {
57 return GEAR_APP_NO_INPUT;
58 }
59
60 // Don't switch if the button is blanked out
61 u16 val = 0;
62 if (DoesPixelAtScreenXYMatchPtrVal(pokegearApp->bgConfig, GF_BG_LYR_MAIN_0, gSystem.touchX, gSystem.touchY, &val) == 1) {
63 return GEAR_APP_NO_INPUT;
64 }
65
66 // Don't switch from an app to itself
67 if (newApp == pokegearApp->app) {
68 return GEAR_APP_NO_INPUT;
69 }
70
71 // Don't switch if the card isn't unlocked
72 if ((newApp == GEAR_APP_RADIO && !(pokegearApp->registeredCards & GEARCARD_RADIO)) || (newApp == GEAR_APP_MAP && !(pokegearApp->registeredCards & GEARCARD_MAP))) {
73 return GEAR_APP_NO_INPUT;
74 }
75 PokegearApp_UpdateAppSwitchButtonBGState(pokegearApp, newApp, 1);
76 PlaySE(newApp != GEAR_APP_CANCEL ? SEQ_SE_GS_GEARAPPLICHANGE : SEQ_SE_GS_GEARCANCEL);
77 pokegearApp->cursorInAppSwitchZone = 0;
78 pokegearApp->menuInputState = MENU_INPUT_STATE_TOUCH;
79 return newApp;
80}

145 RTCDate date;

147 u8 digitBuf[4];

149 GF_RTC_CopyDateTime(&date, &time);
