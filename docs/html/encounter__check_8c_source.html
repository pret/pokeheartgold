564static BOOL FieldSystem_EncounterRateRoll(FieldSystem *fieldSystem, u8 encounterRate, u8 metatileBehavior) {
565 if (encounterRate > 100) {
566 encounterRate = 100;
567 }
568 int avatarState = PlayerAvatar_GetState(fieldSystem->playerAvatar);
569 u8 modifiedRate = 40;
570 if (avatarState != PLAYER_STATE_CYCLING && avatarState != PLAYER_STATE_SURFING && !sub_0205DE98(fieldSystem->playerAvatar)) {
571 modifiedRate -= 20;
572 }
573 if (sub_0205B6F4(metatileBehavior)) {
574 modifiedRate += 40;
575 } else if (avatarState == PLAYER_STATE_CYCLING) {
576 modifiedRate += 30;
577 }
578 modifiedRate += getStepCountEncounterRateBoost(fieldSystem->reverseTurnFrameSteps);
579 if (GetRadioMusicPlayingSeq() == RADIO_MUSIC_SEQ_MARCH) {
580 modifiedRate += 25;
581 } else if (GetRadioMusicPlayingSeq() == RADIO_MUSIC_SEQ_LULLABY) {
582 modifiedRate -= 25;
583 }
584 if (modifiedRate > 100) {
585 modifiedRate = 100;
586 }
587 if (LCRandRange(100) < modifiedRate && FieldSystem_SecondEncounterRoll(fieldSystem, encounterRate)) {
588 return TRUE;
589 }
590
591 return FALSE;
592}

605 return result;

877static BOOL FieldSystem_GenerateRegularEncounter(Pokemon *leadMon, int rodType, EncounterGenState *encounterGen, EncounterSlot *encSlots, u8 encType, int battler, BattleSetup *battleSetup) {
878 u8 slot = 0;
879 u8 level = 0;
880
881 switch (encType) {
882 case ENCOUNTER_TYPE_LAND:
883 if (!EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_LAND, TYPE_STEEL, ABILITY_MAGNET_PULL, &slot) && !EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_LAND, TYPE_ELECTRIC, ABILITY_STATIC, &slot)) {
884 slot = EncounterSlot_WildMonSlotRoll_Land();
885 }
886 slot = ApplyAbilityEffectToSlotLevel(encSlots, NUM_ENCOUNTERS_LAND, encounterGen, slot);
887 level = encSlots[slot].maxLevel;
888 break;
889 case ENCOUNTER_TYPE_ROCK_SMASH:
890 if (!EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_ROCKSMASH, TYPE_STEEL, ABILITY_MAGNET_PULL, &slot) && !EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_ROCKSMASH, TYPE_ELECTRIC, ABILITY_STATIC, &slot)) {
891 slot = EncounterSlot_WildMonSlotRoll_RockSmash();
892 }
893 level = EncounterSlot_WildMonLevelRoll(&encSlots[slot], encounterGen);
894 break;
895 case ENCOUNTER_TYPE_SURFING:
896 if (!EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_SURF, TYPE_STEEL, ABILITY_MAGNET_PULL, &slot) && !EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_SURF, TYPE_ELECTRIC, ABILITY_STATIC, &slot)) {
897 slot = EncounterSlot_WildMonSlotRoll_Surfing();
898 }
899 level = EncounterSlot_WildMonLevelRoll(&encSlots[slot], encounterGen);
900 break;
901 case ENCOUNTER_TYPE_FISHING:
902 if (!EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_FISH, TYPE_STEEL, ABILITY_MAGNET_PULL, &slot) && !EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_FISH, TYPE_ELECTRIC, ABILITY_STATIC, &slot)) {
903 slot = EncounterSlot_WildMonSlotRoll_Fishing(rodType);
904 }
905 level = EncounterSlot_WildMonLevelRoll(&encSlots[slot], encounterGen);
906 break;
907 case ENCOUNTER_TYPE_HEADBUTT:
908 if (!EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_HEADBUTT, TYPE_STEEL, ABILITY_MAGNET_PULL, &slot) && !EncounterSlot_AbilityInfluenceOnSlotChoiceCheck(leadMon, encounterGen, encSlots, NUM_ENCOUNTERS_HEADBUTT, TYPE_ELECTRIC, ABILITY_STATIC, &slot)) {
909 slot = EncounterSlot_WildMonSlotRoll_Headbutt();
910 }
911 level = EncounterSlot_WildMonLevelRoll(&encSlots[slot], encounterGen);
912 break;
913 default:
914 GF_ASSERT(FALSE);
915 }
916
917 if (encSlots[slot].species == SPECIES_UNOWN && !EncounterGen_CanGenerateUnownEncounter(encounterGen)) {
918 return FALSE;
919 }
920 if (DoesAbilitySuppressEncounter(encounterGen, leadMon, level)) {
921 return FALSE;
922 }
923 if (EncounterGen_DoesRepelSuppressEncounter(level, encounterGen) == TRUE) {
924 return FALSE;
925 }
926 generateWildNonShinyAndAddToParty(encSlots[slot].species, level, battler, FALSE, encounterGen, leadMon, battleSetup);
927 return TRUE;
928}
