470static int VoltorbFlipGameState_CalcNextLevel(VoltorbFlipGameState *game) {
471 int i;
472 u32 boardId;
473 VoltorbFlipRoundOutcome roundOutcome;
474
475 RoundSummary *prevRound = VoltorbFlipGameState_GetBoardHistoryTop(game);
476 roundOutcome = prevRound->roundOutcome;
477
478 if (roundOutcome == ROUND_OUTCOME_WON && LEVEL_AT_LEAST(prevRound->boardId, 8)) {
479 return 0; // Lv. 8
480 }
481
482 boardId = prevRound->boardId;
483 // You can reach Lv. 8 if you're at Lv. 5 or higher now and if in each of
484 // the last 5 rounds you:
485 // - Did not lose
486 // - Flipped at least 8 cards
487 if (LEVEL_AT_LEAST(boardId, 5)) {
488 for (i = 0; i < 5; i++) {
489 RoundSummary *round = &game->boardHistory[i];
490 if (round->cardsFlipped < 8 || round->roundOutcome == ROUND_OUTCOME_LOST) {
491 break;
492 }
493 }
494 if (i == 5) {
495 return 0; // Lv. 8
496 }
497 }
498
499 if ((LEVEL_AT_LEAST(boardId, 7) && prevRound->cardsFlipped >= 7) || (LEVEL_AT_LEAST(boardId, 6) && roundOutcome == ROUND_OUTCOME_WON)) {
500 return 1; // Lv. 7
501 }
502 if ((LEVEL_AT_LEAST(boardId, 6) && prevRound->cardsFlipped >= 6) || (LEVEL_AT_LEAST(boardId, 5) && roundOutcome == ROUND_OUTCOME_WON)) {
503 return 2; // Lv. 6
504 }
505 if ((LEVEL_AT_LEAST(boardId, 5) && prevRound->cardsFlipped >= 5) || (LEVEL_AT_LEAST(boardId, 4) && roundOutcome == ROUND_OUTCOME_WON)) {
506 return 3; // Lv. 5
507 }
508 if ((LEVEL_AT_LEAST(boardId, 4) && prevRound->cardsFlipped >= 4) || (LEVEL_AT_LEAST(boardId, 3) && roundOutcome == ROUND_OUTCOME_WON)) {
509 return 4; // Lv. 4
510 }
511 if ((LEVEL_AT_LEAST(boardId, 3) && prevRound->cardsFlipped >= 3) || (LEVEL_AT_LEAST(boardId, 2) && roundOutcome == ROUND_OUTCOME_WON)) {
512 return 5; // Lv. 3
513 }
514 if ((LEVEL_AT_LEAST(boardId, 2) && prevRound->cardsFlipped >= 2) || (roundOutcome == ROUND_OUTCOME_WON)) {
515 return 6; // Lv. 2
516 }
517 return 7; // Lv. 1
518}

608static void VoltorbFlipGameState_PlaceCardsOnBoard(VoltorbFlipGameState *game, CardType type, int n, BOOL isNot1Card) {
609 u8 cardId;
610 int attempts = 0;
611
612 int i = 0;
613 while (i < n) {
614 if (isNot1Card) {
615 cardId = (u32)MTRandom() % 25;
616 } else {
617 cardId = i;
618 }
619 Card *card = VoltorbFlipGameState_GetCard(game, cardId);
620 if (card->type == CARD_TYPE_ONE || !isNot1Card) {
621 card->type = type;
622 } else {
623 // If you can't place the tile on the board within 100 attempts,
624 // skip it.
625 attempts++;
626 if (attempts >= 100) {
627 break;
628 }
629
630 i--;
631 }
632 i++;
633 }

636static BOOL VoltorbFlipGameState_RetryBoardGen(VoltorbFlipGameState *game) {
637 int i;
638 const BoardConfig *config;
639
640 u8 freeMultipliersPerCol[5] = { 0 };
641 u8 freeMultipliersPerRow[5] = { 0 };
642
643 // Multiplier cards (x2 and x3) that you can get at no risk of detonating a
644 // Voltorb.
645 u8 freeMultipliers = 0;
646
647 config = &sBoardConfigs[game->boardId];
648
649 for (i = 0; i < 25; i++) {
650 Card *card = VoltorbFlipGameState_GetCard(game, (u8)i);
651 if (IS_MULTIPLIER_CARD(card->type)) {
652 int col = i % 5;
653 int row = i / 5;
654 int voltorbsInCol = VoltorbFlipGameState_GetVoltorbsAlongAxis(game, AXIS_COL, col);
655 int voltorbsInRow = VoltorbFlipGameState_GetVoltorbsAlongAxis(game, AXIS_ROW, row);
656 if (voltorbsInRow == 0 || voltorbsInCol == 0) {
657 freeMultipliersPerCol[col]++;
658 freeMultipliersPerRow[row]++;
659 freeMultipliers++;
660 }
661 }
662 }
663
664 if (config->maxFreeMultipliers <= freeMultipliers) {
665 return TRUE;
666 }
667
668 for (i = 0; i < 5; i++) {
669 if (config->maxFreeMultipliersPerRowCol <= freeMultipliersPerCol[i] || config->maxFreeMultipliersPerRowCol <= freeMultipliersPerRow[i]) {
670 return TRUE;
671 }
672 }
673
674 return FALSE;
675}

677static void VoltorbFlipGameState_GenerateBoard(VoltorbFlipGameState *game) {
678 GF_ASSERT(game->boardId < 80);
679
680 int voltorbs = sBoardConfigs[game->boardId].voltorbs;
681 int twos = sBoardConfigs[game->boardId].twos;
682 int threes = sBoardConfigs[game->boardId].threes;
683
684 for (int i = 0; i < 1000; i++) {
685 VoltorbFlipGameState_PlaceCardsOnBoard(game, CARD_TYPE_ONE, 25, FALSE);
686 VoltorbFlipGameState_PlaceCardsOnBoard(game, CARD_TYPE_VOLTORB, voltorbs, TRUE);
687 VoltorbFlipGameState_PlaceCardsOnBoard(game, CARD_TYPE_TWO, twos, TRUE);
688 VoltorbFlipGameState_PlaceCardsOnBoard(game, CARD_TYPE_THREE, threes, TRUE);
689 VoltorbFlipGameState_CountVoltorbsInRowCols(game);
690
691 if (!VoltorbFlipGameState_RetryBoardGen(game)) {
692 break;
693 }
694 }
