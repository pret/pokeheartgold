722static void ObjCharTransfer_ReserveTransferBlocks(u32 blockStart, u32 numBlocks, u8 *buffer) {
723 if (buffer != NULL) {
724 u32 blocksMax = ObjCharTransfer_GetBlocksCountFromBufferPtr(buffer);
725 u32 blockEnd = blockStart + numBlocks;
726 for (int i = blockStart; i < blockEnd; ++i) {
727 if (i >= blocksMax) {
728 break;
729 }
730 u32 byteIndex;
731 u8 bitIndex;
732 ObjCharTransfer_BitPositionToByteAndBitIndexPair(i, &byteIndex, &bitIndex);
733 GF_ASSERT(!(buffer[byteIndex] & (1 << bitIndex)));
734 u8 mask = 1 << bitIndex;
735 buffer[byteIndex] |= mask;
736 }
737 }

740static u32 ObjCharTransfer_FindAvailableTransferBlock(u32 reserveSize, u8 *buffer) {
741 if (buffer != NULL) {
742 u32 blocksMax = ObjCharTransfer_GetBlocksCountFromBufferPtr(buffer);
743 for (int i = 0; i < blocksMax; ++i) {
744 u32 bitIndex;
745 u8 byteIndex;
746 ObjCharTransfer_BitPositionToByteAndBitIndexPair(i, &bitIndex, &byteIndex);
747 u8 mask = 1 << byteIndex;
748 int j = 0;
749 while ((buffer[bitIndex] & mask) == 0 && j <= reserveSize) {
750 int num = i + j;
751 ObjCharTransfer_BitPositionToByteAndBitIndexPair(num, &bitIndex, &byteIndex);
752 mask = 1 << byteIndex;
753 if (num >= blocksMax) {
754 return -1;
755 }
756 ++j;
757 }
758 if (j > reserveSize) {
759 return i;
760 }
761 i += j;
762 }
763 }
764 return -1;
765}
