949static u16 GetTileMapIndexFromCoords(u8 x, u8 y, u8 size, u8 mode) {
950 u16 ret;
951 switch (size) {
952 case GF_BG_SCR_SIZE_128x128:
953 GF_ASSERT(x < 16);
954 GF_ASSERT(y < 16);
955 ret = y * 16 + x;
956 break;
957 case GF_BG_SCR_SIZE_256x256:
958 GF_ASSERT(x < 32);
959 GF_ASSERT(y < 32);
960 ret = y * 32 + x;
961 break;
962 case GF_BG_SCR_SIZE_256x512:
963 GF_ASSERT(x < 32);
964 GF_ASSERT(y < 64);
965 ret = y * 32 + x;
966 break;
967 case GF_BG_SCR_SIZE_512x256:
968 GF_ASSERT(x < 64);
969 GF_ASSERT(y < 32);
970 ret = ((x >> 5) * 32 + y) * 32 + (x & 0x1F);
971 break;
972 case GF_BG_SCR_SIZE_512x512:
973 GF_ASSERT(x < 64);
974 GF_ASSERT(y < 64);
975 if (mode == GF_BG_TYPE_TEXT) {
976 ret = (x >> 5) + (y >> 5) * 2;
977 ret *= 1024;
978 ret += (y & 0x1F) * 32 + (x & 0x1F);
979 } else {
980 ret = x + 64 * y;
981 }
982 break;
983 case GF_BG_SCR_SIZE_1024x1024:
984 GF_ASSERT(x < 128);
985 GF_ASSERT(y < 128);
986 ret = y * 128 + x;
987 break;
988 }
989
990 return ret;
991}

1933void CopyGlyphToWindow(Window *window, u8 *glyphPixels, u16 srcWidth, u16 srcHeight, u16 destX, u16 destY, u16 table) {
1934 u8 *windowPixels;
1935 u16 destWidth, destHeight;
1936 int srcRight, srcBottom;
1937 u8 glyphSizeParam;
1938
1939 windowPixels = (u8 *)window->pixelBuffer;
1940 destWidth = (u16)(window->width * 8);
1941 destHeight = (u16)(window->height * 8);
1942
1943 // Don't overflow the window
1944 if (destWidth - destX < srcWidth) {
1945 srcRight = destWidth - destX;
1946 } else {
1947 srcRight = srcWidth;
1948 }
1949 if (destHeight - destY < srcHeight) {
1950 srcBottom = destHeight - destY;
1951 } else {
1952 srcBottom = srcHeight;
1953 }
1954
1955 // Get the max glyph dimensions
1956 // Default: 1x1
1957 glyphSizeParam = 0;
1958 if (srcRight > 8) {
1959 glyphSizeParam |= 1; // 2 wide
1960 }
1961 if (srcBottom > 8) {
1962 glyphSizeParam |= 2; // 2 high
1963 }
1964
1965 if (window->colorMode == GF_BG_CLR_4BPP) {
1966 switch (glyphSizeParam) {
1967 case 0: // 1x1
1968 GLYPH_COPY_4BPP(glyphPixels, 0, 0, srcRight, srcBottom, windowPixels, destX, destY, ConvertPixelsToTiles(destWidth), table);
1969 return;
1970
1971 case 1: // 2x1

1979 return;
1980
1981 case 3: // 2x2

1983 GLYPH_COPY_4BPP(glyphPixels, 8, 0, srcRight - 8, 8, windowPixels, destX, destY, ConvertPixelsToTiles(destWidth), table);
1984 GLYPH_COPY_4BPP(glyphPixels, 0, 8, 8, srcBottom - 8, windowPixels, destX, destY, ConvertPixelsToTiles(destWidth), table);
1985 GLYPH_COPY_4BPP(glyphPixels, 8, 8, srcRight - 8, srcBottom - 8, windowPixels, destX, destY, ConvertPixelsToTiles(destWidth), table);
1986 return;
1987 }
1988 } else { // 8bpp
1989 u8 *convertedSrc;

1999 break;
2000
2001 case 2: // 1x2

2003 GLYPH_COPY_8BPP(convertedSrc, 0, 8, srcRight, srcBottom - 8, windowPixels, destX, destY, ConvertPixelsToTiles(destWidth), table);
2004 break;
2005

2007 GLYPH_COPY_8BPP(convertedSrc, 0, 0, 8, 8, windowPixels, destX, destY, ConvertPixelsToTiles(destWidth), table);
2008 GLYPH_COPY_8BPP(convertedSrc, 8, 0, srcRight - 8, 8, windowPixels, destX, destY, ConvertPixelsToTiles(destWidth), table);

2034 size = window->height * window->width * TILE_SIZE_4BPP;

2036

2040 y0 = y;
2041 for (j = 0; j < 8; j++) {

2047 *(u32 *)(pixelBuffer + y1) = fillWord;

2049 y0++;
2050 }
2051 }
2052 break;
2053 case 1: // down
2054 pixelBuffer += size - 4;
2055 for (i = 0; i < size; i += TILE_SIZE_4BPP) {
2056 y0 = y;
2057 for (j = 0; j < 8; j++) {
2058 y1 = i + (j << 2);

2060 if (y2 < size) {
2061 *(u32 *)(pixelBuffer - y1) = *(u32 *)(pixelBuffer - y2);
2062 } else {
2063 *(u32 *)(pixelBuffer - y1) = fillWord;

2065 y0++;
2066 }
2067 }

2069 case 2: // left
2070 case 3: // right
