110static ItemSlot *Pocket_GetItemSlotForAdd(ItemSlot *slots, u32 count, u16 itemId, u16 quantity, u16 maxQuantity) {
111 s32 i;
112 s32 found = -1;
113 for (i = 0; i < count; i++) {
114 if (slots[i].id == itemId) {
115 // Only one stack allowed per item.
116 // If the resulting stack would be too large,
117 // pretend there's no room for it.
118 if (quantity + slots[i].quantity > maxQuantity) {
119 return NULL;
120 }
121 return &slots[i];
122 }
123 if (found == -1 && slots[i].id == ITEM_NONE && slots[i].quantity == 0) {
124 found = i;
125 }
126 }
127 if (found == -1) {
128 return NULL;
129 }
130 return &slots[found];
131}

285 for (s32 i = 0; i < count - 1; i++) {

294static void SortPocket(ItemSlot *slots, u32 count) {
295 for (s32 i = 0; i < count - 1; i++) {
296 for (s32 j = i + 1; j < count; j++) {
297 if (slots[i].quantity == 0 || (slots[j].quantity != 0 && slots[i].id > slots[j].id)) {
298 SwapItemSlots(&slots[i], &slots[j]);
