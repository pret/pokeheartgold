70 OSThread *next = queue->head;
71 while (next && next->priority <= thread->priority) {
72 if (next == thread) {

74 }
75 next = next->link.next;
76 }
77 if (!next) {

185 if (OSi_RescheduleCount <= 0) {
186 OSThreadInfo *info = &OSi_ThreadInfo;
187 if (info->irqDepth > 0 || OS_GetProcMode() == OS_PROCMODE_IRQ) {
188 info->isNeedRescheduling = TRUE;
189 }
190 else {
191 OSThread *currentThread, *nextThread;
192 currentThread = OSi_GetCurrentThread();
193 nextThread = OS_SelectThread();

195 return;
196 }
197 if (currentThread->state != OS_THREAD_STATE_TERMINATED
198 && OS_SaveContext(&currentThread->context)) {
199 return;
200 }
201 if (OSi_SystemCallbackInSwitchThread) {

206 ((OSSwitchThreadCallback)info->switchCallback) (currentThread, nextThread);
207 }

467void OS_YieldThread(void) {
468 OSThread *current = OS_GetCurrentThread();
469 OSThread *pre = NULL;
470 OSThread *lastThread = NULL;
471 int samePriorityThread = 0;
472 OSIntrMode enable = OS_DisableInterrupts();
473 OSThread *t = OSi_ThreadInfo.list;
474 OSThread *tPre = NULL;
475 while (t) {
476 if (t == current) {
477 pre = tPre;
478 }
479 if (current->priority == t->priority) {
480 lastThread = t;
481 samePriorityThread++;
482 }
483 tPre = t;
484 t = t->next;
485 }
486 if (samePriorityThread <= 1 || lastThread == current) {
487 OS_RestoreInterrupts(enable);
488 return;
489 }
490 if (!pre) {
491 OSi_ThreadInfo.list = current->next;
492 }
493 else {
494 pre->next = current->next;
495 }
496 current->next = lastThread->next;
497 lastThread->next = current;
498 OSi_RescheduleThread();
499 OS_RestoreInterrupts(enable);
500}
