32 MessageFormat *ret;

38 if (ret != NULL) {
39 ret->count = nstr;

42 if (ret->buffer != NULL) {
43 ret->fields = Heap_AllocAtEnd(heapID, nstr * sizeof(MessageFormatFields));
44 if (ret->fields != NULL) {
45 for (i = 0; i < nstr; i++) {
46 MessageFormat_InitFields(&ret->fields[i]);
47 ret->fields[i].msg = String_New(len, heapID);

49 break;
50 }
51 }
52 if (i == nstr) {
53 return ret;
54 }
55 }
56 }
57 }
