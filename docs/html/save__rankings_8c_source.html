127static u32 *Save_RankingSys_GetPlayerStats(SaveData *saveData, enum HeapID heapID) {
128 int i;
129 u32 val;
130 GameStats *gameStats;
131 FrontierSave *frontierSave;
132 u32 *ret;
133
134 gameStats = Save_GameStats_Get(saveData);
135 frontierSave = Save_Frontier_GetStatic(saveData);
136 ret = Heap_AllocAtEnd(heapID, RANKINGS_COUNT * sizeof(u32));
137
138 for (i = 0; i < RANKINGS_COUNT; ++i) {
139 switch (i) {
140 case RANKINGS_RECORD_BATTLE_TOWER_AVG_WIN_STREAK: {
141 val = GameStats_GetCapped(gameStats, GAME_STAT_BATTLE_TOWER_BATTLE_COUNT);
142 if (val != 0) {
143 val = GameStats_GetCapped(gameStats, GAME_STAT_BATTLE_TOWER_WIN_COUNT) / val;
144 }
145 ret[i] = val;
146 break;
147 }
148 case RANKINGS_RECORD_CONTEST_WIN: {
149 val = GameStats_GetCapped(gameStats, GAME_STAT_LOCAL_CONTEST_WINS);
150 val += GameStats_GetCapped(gameStats, GAME_STAT_COMM_CONTEST_WINS);
151 ret[i] = val;
152 break;
153 }
154 case RANKINGS_RECORD_CONTEST_ENTRY: {
155 val = GameStats_GetCapped(gameStats, GAME_STAT_LOCAL_CONTEST_ENTRIES);
156 val += GameStats_GetCapped(gameStats, GAME_STAT_COMM_CONTEST_ENTRIES);
157 if (val != 0) {
158 val = ret[RANKINGS_RECORD_CONTEST_WIN] * 100 / val;
159 }
160 ret[i] = val;
161 break;
162 }
163 default:
164 if (i >= RANKINGS_RECORD_BATTLE_TOWER_SINGLE_WINS && i <= RANKINGS_RECORD_BATTLE_TOWER_WIFI_WINS) {
165 ret[i] = FrontierSave_GetStat(frontierSave, sStatIDs[i], 0xFF);
166 } else {
167 ret[i] = GameStats_GetCapped(gameStats, sStatIDs[i]);
168 }
169 break;
170 }
171 }
172 SaveSubstruct_UpdateCRC(SAVE_RANKINGS);
173 return ret;
174}

226static void SaveRankings_GetSorted(SaveRankings *saveRankings, SaveRankingsSortBuffer *sortBuffer, int groupId, u8 statIdx, u8 scope, SaveRankingsEntry **filteredEntries, u8 filteredEntriesCnt, enum HeapID unused) {
227 SaveRankingsEntry *saveRecordsPtr;
228 int i;
229 int j;
230
231 MI_CpuClear8(sortBuffer, sizeof(SaveRankingsSortBuffer));
232 if (scope == RANKINGS_SCOPE_GLOBAL) {
233 saveRecordsPtr = saveRankings->array[statIdx];
234 MI_CpuCopy8(saveRecordsPtr, sortBuffer->fromSave, RANKINGS_PER_STAT * sizeof(SaveRankingsEntry));
235 for (i = 0; i < filteredEntriesCnt; ++i) {
236 sortBuffer->rankings[sortBuffer->rankingsFromMixingCnt++] = &filteredEntries[i][statIdx];
237 }
238 for (i = 0; i < RANKINGS_PER_STAT; ++i) {

241 }
242 }

244 saveRecordsPtr = saveRankings->array[statIdx + RANKINGS_COUNT];
245 MI_CpuCopy8(saveRecordsPtr, sortBuffer->fromSave, 6 * sizeof(SaveRankingsEntry));
246 for (i = 0; i < filteredEntriesCnt; ++i) {
247 if (filteredEntries[i][statIdx].groupId != 0 && groupId == filteredEntries[i][statIdx].groupId) {
248 sortBuffer->rankings[sortBuffer->rankingsFromMixingCnt++] = &filteredEntries[i][statIdx];
249 }
250 }
251 for (i = 0; i < RANKINGS_PER_STAT; ++i) {
252 if (SaveRankingsEntry_IsInit(&sortBuffer->fromSave[i]) && sortBuffer->fromSave[i].groupId == groupId && !SaveRankingsSortBuffer_EntryAlreadyExists(sortBuffer, &sortBuffer->fromSave[i])) {
253 sortBuffer->rankingsFromSave[sortBuffer->rankingsFromSaveCnt++] = &sortBuffer->fromSave[i];
254 }
255 }
256 }
257 sortBuffer->rankingsTotalCount = sortBuffer->rankingsFromMixingCnt + sortBuffer->rankingsFromSaveCnt;
258 for (i = 0; i < sortBuffer->rankingsFromSaveCnt; ++i) {
259 sortBuffer->rankings[i + sortBuffer->rankingsFromMixingCnt] = sortBuffer->rankingsFromSave[i];
260 }
261 // Bubble-sort
262 for (i = 0; i < sortBuffer->rankingsTotalCount - 1; ++i) {
263 for (j = sortBuffer->rankingsTotalCount - 1; j > i; --j) {
264 if (sortBuffer->rankings[i]->stat < sortBuffer->rankings[j]->stat) {
265 SaveRankingsEntry *tmp = sortBuffer->rankings[i];
266 sortBuffer->rankings[i] = sortBuffer->rankings[j];
267 sortBuffer->rankings[j] = tmp;
268 }
269 }
270 }
271 SaveRankingsEntry_InitArrayOf6(saveRecordsPtr);
272 for (i = 0; i < RANKINGS_PER_STAT && i < sortBuffer->rankingsTotalCount; ++i) {
273 saveRecordsPtr[i] = *sortBuffer->rankings[i];
274 }
275}
