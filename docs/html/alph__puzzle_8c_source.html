817s32 AlphPuzzle_TrySelectTileTouchScreen(AlphPuzzleData *data, u16 touchX, u16 touchY) {
818 u16 x = touchX;
819 u16 y = touchY;
820
821 if (x < 32 || x >= 224) {
822 return -1;
823 }
824
825 x = (x - 32) / 32;
826 y = y / 32;
827
828 for (s32 i = 0; i < 16; i++) {
829 if (data->tileGrid[i].x == x && data->tileGrid[i].y == y) {
830 if (data->tileGrid[i].isImmovable) {
831 return -1;
832 }
833 return i;
834 }
835 }
836
837 return -1;
838}

840// Returns TRUE if the coords are updated,
841// FALSE otherwise.
842// Note: This return value goes unused in
843// code. Instead, it always checks whether
844// the coords actually updated.
845static BOOL AlphPuzzle_GetTileDropCoords(AlphPuzzleData *data, u8 *xOut, u8 *yOut) {
846 int i;
847
848 *xOut = 0;
849 *yOut = 0;
850
851 s16 x = data->tileHoverPixelX;
852 u16 x2;
853 u16 y2;
854
855 // x coords out of bound
856 if (x < 32 || x >= 224) {
857 return FALSE;
858 }
859
860 // no y bounds check because the screen height takes care of that
861
862 x2 = (x - 32) / 32;
863 y2 = data->tileHoverPixelY / 32;
864
865 // Can't place on a corner
866 for (i = 0; i < 4; i++) {
867 if (x2 == sCornerCoords[i][0] && y2 == sCornerCoords[i][1]) {
868 return FALSE;
869 }
870 }
871
872 // Can't place on an occupied tile
873 for (i = 0; i < 16; i++) {
874 if (data->selectedTileIndex != i && data->tileGrid[i].x == x2 && data->tileGrid[i].y == y2) {
875 return FALSE;
876 }
877 }
878
879 *xOut = x2;
880 *yOut = y2;
881
882 return TRUE;
883}

885static int AlphPuzzleMainSeq_HoldingTile_impl(AlphPuzzleData *data) {
886 if (!System_GetTouchHeld()) {
887 AlphPuzzle_ClearOrSetBgTilesAtCoords(data, data->selectedTile->x, data->selectedTile->y, 0);
888 AlphPuzzle_ClearOrSetBgTilesAtCoords(data, data->tileHoverTileX, data->tileHoverTileY, 0);
889 ScheduleBgTilemapBufferTransfer(data->bgConfig, 2);
890 if (data->tileHoverTileX == 0 && data->tileHoverTileY == 0) {
891 PlaySE(SEQ_SE_DP_BOX01);
892 AlphPuzzle_PlaceTileInGrid(data, data->selectedTileIndex, data->selectedTile->x, data->selectedTile->y, data->selectedTile->rotation);
893 } else {
894 PlaySE(SEQ_SE_GS_SEKIBAN_SENTAKU); // SE Slate Select
895 AlphPuzzle_PlaceTileInGrid(data, data->selectedTileIndex, data->tileHoverTileX, data->tileHoverTileY, data->selectedTile->rotation);
896 }
897 AlphPuzzle_UpdateSelectedTile(data, -1, FALSE);
898 if (AlphPuzzle_CheckComplete(data)) {
899 return ALPH_PUZZLE_STATE_CLEAR;
900 }
901 return ALPH_PUZZLE_STATE_WAIT_FOR_INPUT;
902 }
903
904 s16 x = gSystem.touchX;
905 s16 y = gSystem.touchY;
906
907 data->tileLastHoverPixelX = data->tileHoverPixelX;
908 data->tileLastHoverPixelY = data->tileHoverPixelY;
909
910 if (y < 16) {
911 y = 16;
912 } else if (y > 176) {
913 y = 176;
914 }
915
916 if (x < 48) {
917 x = 48;
918 } else if (x > 208) {
919 x = 208;
920 }
921
922 Sprite_SetPositionXY(data->sprites[ALPH_SPRITE_INDEX_PREV_CURSOR], x, y);
923 AlphPuzzle_SetSpritePosition_HandleRotation(data->selectedTile, x - 2, y - 2);
924
925 u8 xOut;
926 u8 yOut;
927
928 data->tileHoverPixelX = x;
929 data->tileHoverPixelY = y;
930
931 AlphPuzzle_GetTileDropCoords(data, &xOut, &yOut);
932
933 if (data->tileHoverTileX != xOut || data->tileHoverTileY != yOut) {
934 AlphPuzzle_UpdateHoverGraphicOnBg(data, xOut, yOut);
935 data->tileHoverTileX = xOut;
936 data->tileHoverTileY = yOut;
937 }
938
939 return ALPH_PUZZLE_STATE_HOLDING_TILE;
940}
