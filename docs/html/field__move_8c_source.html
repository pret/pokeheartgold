120void FieldMove_InitCheckData(FieldSystem *fieldSystem, FieldMoveCheckData *checkData) {
121 checkData->fieldSystem = fieldSystem;
122 checkData->mapId = fieldSystem->location->mapId;
123 checkData->flag = 0;
124
125 LocalMapObject *facingObject;
126
127 FieldSystem_GetFacingObject(fieldSystem, &facingObject);
128 checkData->facingObject = facingObject;
129
130 if (facingObject != NULL) {
131 switch (MapObject_GetSpriteID(facingObject)) {
132 case SPRITE_ROCK:
133 checkData->flag |= FIELD_MOVE_CHECK_ROCK;
134 break;
135 case SPRITE_BREAKROCK:
136 checkData->flag |= FIELD_MOVE_CHECK_BREAKROCK;
137 break;
138 case SPRITE_TREE:
139 checkData->flag |= FIELD_MOVE_CHECK_TREE;
140 break;
141 }
142 }
143
144 u32 x = PlayerAvatar_GetXCoord(fieldSystem->playerAvatar);
145 u32 z = PlayerAvatar_GetZCoord(fieldSystem->playerAvatar);
146 u32 standingTile = GetMetatileBehavior(fieldSystem, x, z);
147
148 PlayerAvatar_GetCoordsInFront(fieldSystem->playerAvatar, &x, &z);
149 u32 facingTile = GetMetatileBehavior(fieldSystem, x, z);
150
151 if (Field_PlayerCanSurfOnTile(fieldSystem->playerAvatar, standingTile, facingTile)) {
152 checkData->flag |= FIELD_MOVE_CHECK_WATER;
153 }
154 if (MetatileBehavior_IsRockClimbInDirection(facingTile, PlayerAvatar_GetFacingDirection(fieldSystem->playerAvatar))) {
155 checkData->flag |= FIELD_MOVE_CHECK_ROCKCLIMB;
156 }
157 if (MetatileBehavior_IsWaterfall(facingTile)) {
158 checkData->flag |= FIELD_MOVE_CHECK_WATERFALL;
159 }
160 if (MetatileBehavior_IsWhirlpool(facingTile)) {
161 checkData->flag |= FIELD_MOVE_CHECK_WHIRLPOOL;
162 }
163 if (LocalFieldData_GetWeatherType(Save_LocalFieldData_Get(fieldSystem->saveData)) == WEATHER_FLASH) {
164 checkData->flag |= FIELD_MOVE_CHECK_FLASH;
165 }
166 if (MetatileBehavior_IsHeadbutt(facingTile) || FieldSystem_FacingModelIsHeadbuttTree(fieldSystem)) {
167 checkData->flag |= FIELD_MOVE_CHECK_HEADBUTT;
168 }
169}
