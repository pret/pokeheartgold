113static BOOL Task_TryHeadbuttEncounter(TaskManager *taskManager) {
114 HeadbuttEncounterData *headbuttTable;
115 FieldSystem *fieldSystem = TaskManager_GetFieldSystem(taskManager);
116 TaskData_TryHeadbuttEncounter *didHeadbuttStartBattle = TaskManager_GetEnvironment(taskManager);
117 headbuttTable = AllocAtEndAndReadWholeNarcMemberByIdPair(NARC_arc_headbutt, fieldSystem->location->mapId, HEAP_ID_FIELD2);
118 if (headbuttTable->numRegularTrees != 0 || headbuttTable->numSecretTrees != 0) {
119 BattleSetup *setup;
120 u32 x;
121 u32 y;
122 GetCoordsOfFacingTree(fieldSystem, &x, &y);
123 u32 trainerId = PlayerProfile_GetTrainerID(Save_PlayerData_GetProfile(fieldSystem->saveData));
124 enum TreeType treeType = (enum TreeType)Headbutt_GetTreeTypeFromTable(headbuttTable->numRegularTrees, headbuttTable->numSecretTrees, trainerId, x, y, headbuttTable->treeCoords);
125 if (treeType == TREETYPE_NONE) {
126 Heap_Free(headbuttTable);
127 Heap_Free(didHeadbuttStartBattle);
128 return TRUE;
129 }
130 HeadbuttSlot *headbuttEncounterSlots;
131 if (treeType == TREETYPE_COMMON) {
132 headbuttEncounterSlots = headbuttTable->common;
133 } else if (treeType == TREETYPE_RARE) {
134 headbuttEncounterSlots = headbuttTable->rare;
135 } else if (treeType == TREETYPE_SECRET) {
136 headbuttEncounterSlots = headbuttTable->secret;
137 } else {
138 GF_ASSERT(FALSE);
139 Heap_Free(headbuttTable);
140 Heap_Free(didHeadbuttStartBattle);
141 return TRUE;
142 }
143 if (FieldSystem_PerformHeadbuttEncounterCheck(fieldSystem, &setup, headbuttEncounterSlots)) {
144 *didHeadbuttStartBattle->resultPtr = TRUE;
145 Heap_Free(headbuttTable);
146 Heap_Free(didHeadbuttStartBattle);
147 FieldSystem_StartForcedWildBattle(fieldSystem, taskManager, setup);
148 return FALSE;
149 }
150 }
151 Heap_Free(headbuttTable);
152 Heap_Free(didHeadbuttStartBattle);
153 return TRUE;
154}

156static s8 Headbutt_GetTreeTypeFromTable(u16 numRegularTrees, u16 numSecretTrees, u32 trainerId, u32 x, u32 y, s16 treeCoords[][2]) {
157 // Based on your trainer ID and the tree you're facing, choose which encounter table applies
158 // Not all trees are headbuttable, this is determined by trainer ID
159 u16 i;
160 u16 numCoordsRegular = numRegularTrees * 6;
161 u16 numCoordsSecret = numSecretTrees * 6;
162 for (i = 0; i < numCoordsRegular; i++) {
163 if (x == treeCoords[i][0] && y == treeCoords[i][1]) {
164 return Headbutt_GetTreeType_Regular(i / 6, numRegularTrees, trainerId);
165 }
166 }
167 // Check whether the facing tree is in the rare table
168 for (i = numCoordsRegular; i < numCoordsRegular + numCoordsSecret; i++) {
169 if (x == treeCoords[i][0] && y == treeCoords[i][1]) {
170 return TREETYPE_SECRET;
171 }
172 }
173 return TREETYPE_NONE;
174}

176static s8 Headbutt_GetTreeType_Regular(u8 whichTree, u8 numTrees, u32 trainerId) {
177 s8 ret = TREETYPE_NONE;
178 u8 trainerIdLastDigit = trainerId % 10;
179 if (numTrees >= 5) {
180 u8 column = whichTree % 5;
181 ret = sRareTreeLUT_5Plus[trainerIdLastDigit][column];
182 } else if (numTrees == 4) {
183 ret = sRareTreeLUT_4[trainerIdLastDigit][whichTree];
184 } else if (numTrees == 3) {
185 ret = sRareTreeLUT_3[trainerIdLastDigit][whichTree];
186 } else if (numTrees == 2) {
187 ret = sRareTreeLUT_2[trainerIdLastDigit][whichTree];
188 } else if (numTrees == 1) {
189 ret = sRareTreeLUT_1[trainerIdLastDigit][0];
190 } else {
191 // numTrees == 0 --> unreachable
192 GF_ASSERT(FALSE);
193 }
194 return ret;
195}

197static void GetCoordsOfFacingTree(FieldSystem *fieldSystem, u32 *x, u32 *y) {
198 int inFrontX, inFrontY;
199 PlayerAvatar_GetCoordsInFront(fieldSystem->playerAvatar, &inFrontX, &inFrontY);
200 if (FollowMon_IsActive(fieldSystem)) {
201 LocalMapObject *object = FollowMon_GetMapObject(fieldSystem);
202 u32 followingMonX = MapObject_GetXCoord(object);
203 u32 followingMonZ = MapObject_GetZCoord(object);
204 if (inFrontX == followingMonX && inFrontY == followingMonZ) {
205 u8 dir = MapObject_GetFacingDirection(object);
206 inFrontX = GetDeltaXByFacingDirection(dir) + followingMonX;
207 inFrontY = GetDeltaYByFacingDirection(dir) + followingMonZ;
208 }
209 }
210 *x = inFrontX;
211 *y = inFrontY;
212}
