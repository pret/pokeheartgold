281u8 Radio_GetTunedStationID(PokegearRadioAppData *radioApp, s16 x, s16 y, u8 *pSignalStrengthRet) {
282 u8 ret;
283 int station = TouchscreenHitbox_FindHitboxAtPoint(sTuningHitboxes[radioApp->stationSelection], x, y);
284 if (station == TOUCH_MENU_NO_INPUT) {
285 ret = 255;
286 if (pSignalStrengthRet != NULL) {
287 *pSignalStrengthRet = 0;
288 }
289 } else {
290 switch (radioApp->stationSelection) {
291 case RADIO_STATION_SELECTION_JOHTO:
292 case RADIO_STATION_SELECTION_KANTO:
293 case RADIO_STATION_SELECTION_KANTO_EXPN:
294 case RADIO_STATION_SELECTION_NO_SIGNAL:
295 ret = station / 2u;
296 break;
297 case RADIO_STATION_SELECTION_ROCKET:
298 ret = 6;
299 break;
300 case RADIO_STATION_SELECTION_MAHOGANY:
301 ret = 7;
302 break;
303 default:
304 ret = 5;
305 break;
306 }
307 if (pSignalStrengthRet != NULL) {
308 // 1: yes static
309 // 2: no static
310 *pSignalStrengthRet = 2 - (station % 2u);
311 }
312 }
313 return ret;

317 int ret;
