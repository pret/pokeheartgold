584static void *loadResourceFromNarc(NARC *narc, int fileId, BOOL compressed, enum HeapID heapID, BOOL atEnd) {
585 void *data = NARC_AllocAndReadWholeMember(narc, fileId, heapID);
586 if (data != NULL) {
587 if (compressed) {
588 void *uncompData;
589 if (atEnd == 0) { // explicit comparison to 0 is required to match
590 uncompData = Heap_Alloc(heapID, MI_GetUncompressedSize(data));
591 } else {
592 uncompData = Heap_AllocAtEnd(heapID, MI_GetUncompressedSize(data));
593 }
594 if (uncompData != NULL) {
595 MI_UncompressLZ8(data, uncompData);
596 Heap_Free(data);
597 }
598 // Possible memory leak if uncompData == NULL
599 data = uncompData;
600 }
601 }
602 return data;
603}
