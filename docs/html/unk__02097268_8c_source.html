263static u8 getObjectLevelBoost(u8 days, u8 objectType) {
264 if (objectType == SAFARI_ZONE_OBJECT_TYPE_NONE) {
265 return 0;
266 }
267 for (int i = 0; i < 5; ++i) {
268 if (days < sObjectLevelBoosts[objectType - 1][i]) {
269 return i + 1;
270 }
271 }
272 if (days < 25) {
273 return 6;
274 }
275 return 7;
