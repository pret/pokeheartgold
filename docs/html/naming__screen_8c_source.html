538BOOL NamingScreenApp_Main(OverlayManager *ovyMan, int *pState) {
539 NamingScreenAppData *data = OverlayManager_GetData(ovyMan);
540
541 switch ((NamingScreenMainState)*pState) {
542 case NS_MAIN_STATE_WAIT_FADE_IN:
543 if (IsPaletteFadeFinished()) {
544 *pState = NS_MAIN_STATE_FINISH_INIT;
545 data->delayCounter = 0;
546 }
547 break;
548 case NS_MAIN_STATE_FINISH_INIT:
549 ++data->delayCounter;
550 NamingScreen_PlaceCursorSprite(data);
551 NamingScreen_UpdateSpritesAnims(data->spriteAnimUpdateReq, data->uiSprites, data->pageNum);
552 if (data->delayCounter > 5) {
553 *pState = NS_MAIN_STATE_INPUT_LOOP;
554 data->delayCounter = 0;
555 }
556 break;
557 case NS_MAIN_STATE_INPUT_LOOP:
558 switch (data->pageSwitchState) {
559 case NS_PAGESWITCH_STATE_DRAW_NEW_PAGE:
560 case NS_PAGESWITCH_STATE_SET_INIT_POS:
561 case NS_PAGESWITCH_STATE_MOVE_BGS:
562 case NS_PAGESWITCH_STATE_FINISH_BGS:
563 break;
564 case NS_PAGESWITCH_STATE_IDLE:
565 if (!data->kbCursor.ignoreInput) {
566 *pState = NamingScreen_HandleInput(data, (NamingScreenMainState)*pState);
567 }
568 NamingScreen_PlaceCursorSprite(data);
569 break;
570 case NS_PAGESWITCH_STATE_PRINT_BATTLE_MESSAGE:
571 NamingScreen_PrepareBattleMessage(data, ovyMan);
572 FillWindowPixelBuffer(&data->windows[9], 15);
573 DrawFrameAndWindow2(&data->windows[9], FALSE, 256, 10);
574 data->battleMessageTextPrinterId = AddTextPrinterParameterized(&data->windows[9], 1, data->battleMsgString, 0, 0, 1, NULL);
575 CopyWindowToVram(&data->windows[9]);
576 data->pageSwitchState = NS_PAGESWITCH_STATE_WAIT_BATTLE_MESSAGE;
577 break;
578 case NS_PAGESWITCH_STATE_WAIT_BATTLE_MESSAGE:
579 if (!TextPrinterCheckActive(data->battleMessageTextPrinterId)) {
580 PlaySE(SEQ_SE_DP_PIRORIRO);
581 ++data->spriteAnimUpdateReq[6];
582 data->delayCounter = 0;
583 data->pageSwitchState = NS_PAGESWITCH_STATE_DELAY_AND_FADE_OUT;
584 }
585 break;
586 case NS_PAGESWITCH_STATE_DELAY_AND_FADE_OUT:
587 ++data->delayCounter;
588 if (data->delayCounter > 30) {
589 BeginNormalPaletteFade(2, 0, 0, RGB_BLACK, 16, 1, HEAP_ID_NAMING_SCREEN);
590 *pState = NS_MAIN_STATE_WAIT_FADE_OUT;
591 }
592 break;
593 }
594 NamingScreen_HandlePageSwitch(data->bgConfig, data->windows, &data->pageSwitchState, data->pageNum, &data->activeKeyboardBgId, data->bgPosVecs, data->uiSprites, data->charData->pRawData);
595 NamingScreen_UpdateSpritesAnims(data->spriteAnimUpdateReq, data->uiSprites, data->pageNum);
596 NamingScreen_PaletteGlowEffect(&data->plttGlowEffectAngle);
597 break;
598 case NS_MAIN_STATE_WAIT_FADE_OUT:
599 if (IsPaletteFadeFinished()) {
600 return TRUE;
601 }
602 break;
603 }
604
605 SpriteList_RenderAndAnimateSprites(data->spriteList);
606 return FALSE;
607}

609static NamingScreenMainState NamingScreen_HandleInput(NamingScreenAppData *data, NamingScreenMainState state) {
610 NamingScreenMainState ret = state;
611
612 NamingScreen_GetPlayerInput(data);
613 if (gSystem.newKeys & PAD_BUTTON_SELECT) {
614 if (!Sprite_GetDrawFlag(data->uiSprites[8])) {
615 Sprite_SetDrawFlag(data->uiSprites[8], TRUE);
616 return ret;
617 }
618 if (data->type != NAME_SCREEN_UNK4) {
619 data->pageSwitchState = NS_PAGESWITCH_STATE_DRAW_NEW_PAGE;
620 ++data->pageNum;
621 if (data->pageNum >= 3) {
622 data->pageNum = 0;
623 }
624 ++data->spriteAnimUpdateReq[data->pageNum];
625 NamingScreen_LoadKeyboardLayout(data->keyboard, data->pageNum);
626 PlaySE(SEQ_SE_DP_SYU03);
627 data->kbCursor.showCursor = TRUE;
628 }
629 ++data->spriteAnimUpdateReq[data->pageNum];
630 NamingScreen_LoadKeyboardLayout(data->keyboard, data->pageNum);
631 PlaySE(SEQ_SE_DP_SYU03);
632 } else if (gSystem.newKeys & PAD_BUTTON_A) {
633 ret = NamingScreen_HandleCharacterInput(data, data->keyboard[data->kbCursor.y][data->kbCursor.x], TRUE);
634 data->kbCursor.showCursor = TRUE;
635 } else if (data->isTouchInput == TRUE) {
636 ret = NamingScreen_HandleCharacterInput(data, data->keyboard[data->kbCursor.y][data->kbCursor.x], FALSE);
637 } else if (gSystem.newKeys & PAD_BUTTON_B) {
638 ret = NamingScreen_HandleCharacterInput(data, NAME_SCREEN_BUTTON_BACK, TRUE);
639 } else if (gSystem.newKeys & PAD_BUTTON_R) {
640 ret = NamingScreen_HandleCharacterInput(data, NAME_SCREEN_BUTTON_PAGE_JP_UNUSED_2, TRUE);
641 }
642 return ret;
643}
