1340static u16 OakSpeech_AdditiveTransformColor(u16 pltt, s8 delta) {
1341 int r = pltt & 0x1F;
1342 int g = (pltt >> 5) & 0x1F;
1343 int b = (pltt >> 10);
1344
1345 r += delta;
1346 g += delta;
1347 b += delta;
1348
1349 if (r > 31) {
1350 r = 31;
1351 } else if (r < 0) {
1352 r = 0;
1353 }
1354 if (g > 31) {
1355 g = 31;
1356 } else if (g < 0) {
1357 g = 0;
1358 }
1359 if (b > 31) {
1360 b = 31;
1361 } else if (b < 0) {
1362 b = 0;
1363 }
1364 return (b << 10) | (g << 5) | r;
1365}
