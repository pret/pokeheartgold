971static inline void runPokepicAnim(u8 *pActive, u8 *pWhichAnimStep, u8 *pWhichAnim, u8 *pStepDelay, u8 *pLoopTimers, const PokepicAnimScript *animScript) {
972 if (*pActive != 0) {
973 if (*pStepDelay == 0) {
974 ++(*pWhichAnim);
975 while (animScript[*pWhichAnim].next < -1) {
976 ++pLoopTimers[*pWhichAnim];
977 if (animScript[*pWhichAnim].duration == pLoopTimers[*pWhichAnim] || animScript[*pWhichAnim].duration == 0) {
978 pLoopTimers[*pWhichAnim] = 0;
979 ++pWhichAnim; // ++(*pWhichAnim);
980 } else {
981 *pWhichAnim = -2 - animScript[*pWhichAnim].next;
982 }
983 }
984 if (animScript[*pWhichAnim].next == -1 || *pWhichAnim >= 10) {
985 *pWhichAnimStep = 0;
986 *pActive = 0;
987 } else {
988 *pWhichAnimStep = animScript[*pWhichAnim].next;
989 *pStepDelay = animScript[*pWhichAnim].duration;
990 }
991 } else {
992 --(*pStepDelay);
993 }
994 }
995}

997static void Pokepic_RunAnimInternal(Pokepic *pokepic) {
998 if (pokepic->animActive != 0) {
999 if (pokepic->animStepDelay == 0) {
1000 ++pokepic->whichAnim;
1001 while (pokepic->animScript[pokepic->whichAnim].next < -1) {
1002 ++pokepic->animLoopTimers[pokepic->whichAnim];
1003 if (pokepic->animScript[pokepic->whichAnim].duration == pokepic->animLoopTimers[pokepic->whichAnim] || pokepic->animScript[pokepic->whichAnim].duration == 0) {
1004 pokepic->animLoopTimers[pokepic->whichAnim] = 0;
1005 ++pokepic->whichAnim;
1006 } else {
1007 pokepic->whichAnim = -2 - pokepic->animScript[pokepic->whichAnim].next;
1008 }
1009 }
1010 if (pokepic->whichAnim >= 10 || pokepic->animScript[pokepic->whichAnim].next == -1) {
1011 pokepic->whichAnimStep = 0;
1012 pokepic->animActive = 0;
1013 pokepic->drawParam.xOffset = 0;
1014 } else {
1015 pokepic->whichAnimStep = pokepic->animScript[pokepic->whichAnim].next;
1016 pokepic->animStepDelay = pokepic->animScript[pokepic->whichAnim].duration;
1017 pokepic->drawParam.xOffset = pokepic->animScript[pokepic->whichAnim].xOffset;
1018 }
1019 } else {
1020 --pokepic->animStepDelay;
1021 }
1022 }
1023}
