236static void RoamerLocationSetRandom(RoamerSaveData *roamer, u8 roamer_idx, u32 last_loc) {
237 u32 roamer_cur_loc, roamer_test_loc;
238 u8 loc_min, loc_num;
239 u8 loc_cur_rand;
240
241 roamer_cur_loc = sRoamerLocations[Roamer_GetLocation(roamer, roamer_idx)];
242 if (roamer_idx == ROAMER_RAIKOU || roamer_idx == ROAMER_ENTEI) {
243 loc_num = ROAMER_LOC_JOHTO_COUNT;
244 loc_min = ROAMER_LOC_JOHTO_START;
245 } else {
246 loc_num = ROAMER_LOC_KANTO_COUNT;
247 loc_min = ROAMER_LOC_KANTO_START;
248 }
249 do {
250 loc_cur_rand = (LCRandom() % loc_num) + loc_min;
251 roamer_test_loc = sRoamerLocations[loc_cur_rand];
252 } while (roamer_test_loc == last_loc || roamer_test_loc == roamer_cur_loc);
253 ApplyRoamerLocation(roamer, roamer_idx, loc_cur_rand, roamer_test_loc);
254}

256static void RoamerLocationUpdateEx(RoamerSaveData *roamer, u8 roamer_idx, u32 last_loc) {
257 u8 roamer_last_loc;
258 u8 roamer_next_loc;
259 u32 roamer_next_mapno;
260 u8 sel_idx;
261 const struct RoamerAdjacency *adj;
262
263 roamer_last_loc = Roamer_GetLocation(roamer, roamer_idx);
264 adj = &sRoamerAdjacencyTable[roamer_last_loc];
265 if (adj->count == 1) {
266 roamer_next_loc = adj->neighbors[0];
267 roamer_next_mapno = sRoamerLocations[roamer_next_loc];
268 if (roamer_next_mapno == last_loc) {
269 RoamerLocationSetRandom(roamer, roamer_idx, last_loc);
270 } else {
271 ApplyRoamerLocation(roamer, roamer_idx, roamer_next_loc, roamer_next_mapno);
272 }
273 } else {
274 while (1) {
275 sel_idx = LCRandRange(adj->count);
276 roamer_next_loc = adj->neighbors[sel_idx];
277 roamer_next_mapno = sRoamerLocations[roamer_next_loc];
278 if (roamer_next_mapno != last_loc) {
279 ApplyRoamerLocation(roamer, roamer_idx, roamer_next_loc, roamer_next_mapno);
280 break;
281 }
282 }
283 }
