516static BOOL SpriteList_LoadCellAndAnimForSprite(SpriteList *spriteList, const SpriteResourcesHeader *resHdr, Sprite *sprite, enum HeapID heapID) {
517 sprite->flag = SpriteResourcesHeader_GetSpriteAnimType(resHdr);
518 sprite->imageProxy = *resHdr->imageProxy;
519 sprite->paletteProxy = *resHdr->plttProxy;
520 Sprite_SetCellBankPtr(resHdr->cellData, sprite);
521 if (resHdr->cellAnim != NULL) {
522 Sprite_SetAnimBankPtr(resHdr->cellAnim, sprite);
523 } else {
524 Sprite_SetAnimBankPtr(spriteList->animBank, sprite);
525 }
526 if (sprite->flag == SPRITE_ANIM_TYPE_MULTICELL) {
527 Sprite_SetMultiCellBankPtr(resHdr->multiCellData, sprite);
528 Sprite_SetMultiAnimBankPtr(resHdr->multiCellAnim, sprite);
529 Sprite_InitMultiCellAnimation(sprite, heapID);
530 } else if (sprite->flag == SPRITE_ANIM_TYPE_CELL_TRANSFER) {
531 Sprite_InitCellAnimationWithTransfer(resHdr, sprite, heapID);
532 } else {
533 Sprite_InitCellAnimation(sprite, heapID);
534 }
535 sprite->priority = resHdr->priority;
536 return TRUE;
537}

656static void SpriteList_InsertSprite(SpriteList *spriteList, Sprite *sprite) {
657 Sprite *head;
658 if (spriteList->dummy.next == &spriteList->dummy) {
659 spriteList->dummy.next = sprite;
660 spriteList->dummy.prev = sprite;
661 sprite->prev = &spriteList->dummy;
662 sprite->next = &spriteList->dummy;
663 return;
664 }
665
666 if (spriteList->dummy.prev->drawPriority <= sprite->drawPriority) {
667 sprite->prev = spriteList->dummy.prev;
668 spriteList->dummy.prev->next = sprite;
669 sprite->next = &spriteList->dummy;
670 spriteList->dummy.prev = sprite;
671 return;
672 }
673
674 head = spriteList->dummy.next;
675 while (head != &spriteList->dummy) {
676 if (head->drawPriority > sprite->drawPriority) {
677 head->prev->next = sprite;
678 sprite->prev = head->prev;
679 head->prev = sprite;
680 sprite->next = head;
681 return;
682 }
683 head = head->next;
684 }
685}
