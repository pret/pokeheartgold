669static u8 ComputeCompatibilityBetweenBoxMons(BoxPokemon **parents) {
670 u16 eggGroups[NUM_DAYCARE_MONS][EGG_GROUPS_PER_MON];
671 u16 species[NUM_DAYCARE_MONS];
672 u32 otIds[NUM_DAYCARE_MONS];
673 u32 genders[NUM_DAYCARE_MONS];
674
675 for (u32 i = 0; i < NUM_DAYCARE_MONS; i++) {
676 species[i] = GetBoxMonData(parents[i], MON_DATA_SPECIES, NULL);
677 otIds[i] = GetBoxMonData(parents[i], MON_DATA_OT_ID, NULL);
678 genders[i] = GetGenderBySpeciesAndPersonality(species[i], GetBoxMonData(parents[i], MON_DATA_PERSONALITY, NULL));
679 eggGroups[i][0] = GetMonBaseStat(species[i], BASE_EGG_GROUP_1);
680 eggGroups[i][1] = GetMonBaseStat(species[i], BASE_EGG_GROUP_2);
681 }
682 // The Undiscovered egg group cannot breed.
683 if (eggGroups[0][0] == EGG_GROUP_UNDISCOVERED || eggGroups[1][0] == EGG_GROUP_UNDISCOVERED) {
684 return PARENTS_INCOMPATIBLE;
685 }
686 // Two Ditto cannot breed.
687 if (eggGroups[0][0] == EGG_GROUP_DITTO && eggGroups[1][0] == EGG_GROUP_DITTO) {
688 return PARENTS_INCOMPATIBLE;
689 }
690 // If one of the two Pokemon is a Ditto, treat them as different species. See below.
691 if (eggGroups[0][0] == EGG_GROUP_DITTO || eggGroups[1][0] == EGG_GROUP_DITTO) {
692 if (otIds[0] == otIds[1]) {
693 return PARENTS_LOW_COMPATIBILITY;
694 } else {
695 return PARENTS_MED_COMPATIBILITY;
696 }
697 }
698 // Same-gender pairs cannot breed.
699 if (genders[0] == genders[1]) {
700 return PARENTS_INCOMPATIBLE;
701 }
702 // Genderless Pokemon cannot breed (without Ditto).
703 if (genders[0] == MON_GENDERLESS || genders[1] == MON_GENDERLESS) {
704 return PARENTS_INCOMPATIBLE;
705 }
706 // Nonmatching egg groups are incompatible.
707 if (!EggGroupsHaveMatch(eggGroups[0], eggGroups[1])) {
708 return PARENTS_INCOMPATIBLE;
709 }
710 // Compatibility is higher if both Pokemon are the same species and/or from two different OTs.
711 if (species[0] == species[1]) {
712 if (otIds[0] != otIds[1]) {
713 return PARENTS_MAX_COMPATIBILITY;
714 } else {
715 return PARENTS_MED_COMPATIBILITY;
716 }
717 } else {
718 if (otIds[0] != otIds[1]) {
719 return PARENTS_MED_COMPATIBILITY;
720 } else {
721 return PARENTS_LOW_COMPATIBILITY;
722 }
723 }
724}

789 Save_Daycare_SetEggCycleCounter(dayCare, 0);
790 u8 to_sub = GetEggCyclesToSubtract(party);
791 for (slot = 0; slot < Party_GetCount(party); slot++) {

794 friendship = GetMonData(mon, MON_DATA_FRIENDSHIP, NULL);
795 if (friendship != 0) {
796 if (friendship >= to_sub) {
797 friendship -= to_sub;
798 } else {
799 friendship -= 1;
800 }
801 SetMonData(mon, MON_DATA_FRIENDSHIP, &friendship);
802 } else {
803 return TRUE;
804 }
805 }
806 }
807 }
808 return FALSE;
