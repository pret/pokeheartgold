674void MoveItemSlotInList(ItemSlot *slots, int from, int to, int pocket, enum HeapID heapID) {
675 ItemSlot *buf;
676 int i, j;
677 if (from == to) {
678 return;
679 }
680
681 buf = Heap_Alloc(heapID, sPocketCounts[pocket] * sizeof(ItemSlot));
682
683 j = 0;
684 for (i = 0; i < sPocketCounts[pocket]; i++) {
685 if (from != i) {
686 if (to == j) {
687 buf[j] = slots[from];
688 j++;
689 }
690 buf[j] = slots[i];
691 j++;
692 }
693 }
694
695 if (to == sPocketCounts[pocket] - 1) {
696 buf[j] = slots[from];
697 }
698
699 for (j = 0; j < sPocketCounts[pocket]; j++) {
700 slots[j] = buf[j];
701 }
702
703 Heap_Free(buf);
704}
