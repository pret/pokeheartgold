567void TintPalette_GrayScale(u16 *palette, int count) {
568 int i, r, g, b;
569 u32 gray;
570 for (i = 0; i < count; ++i) {
571 r = *palette & 0x1F;
572 g = (*palette >> 5) & 0x1F;
573 b = (*palette >> 10) & 0x1F;
574 // 0.3 * red + 0.59 * g + 0.1133 * b
575 gray = (76 * r + 151 * g + 29 * b) >> 8;
576 *palette++ = (gray << 10) | (gray << 5) | gray;
577 }

619static void FadePaletteTowardsColorStep(const u16 *src, u16 *dest, int duration, int step, int rTarget, int gTarget, int bTarget) {
620 int i, r, g, b;
621
622 for (i = 0; i < 16; ++i) {
623 r = *src & 0x1F;
624 g = (*src >> 5) & 0x1F;
625 b = (*src >> 10) & 0x1F;
626
627 r += (rTarget - r) * step / duration;
628 g += (gTarget - g) * step / duration;
629 b += (bTarget - b) * step / duration;
630
631 *dest = (b << 10) | (g << 5) | r;
632
633 ++src;
634 ++dest;
635 }
636}
