179static enum PalParkEncounterType GetEncounterArea(FieldSystem *fieldSystem, int playerX, int playerY) {
180 int behavior = GetMetatileBehavior(fieldSystem, playerX, playerY);
181 int quadrant = (playerX < 32) ? 0 : 1;
182 quadrant += (playerY < 32) ? 0 : 2;

192static BOOL TryStartEncounter(FieldSystem *fieldSystem, CatchingShow *catchingShow, int playerX, int playerY) {
193 int i;
194 int encounterChance, totalRarity = 0;
195 enum PalParkEncounterType area = GetEncounterArea(fieldSystem, playerX, playerY);
196
197 if (area == PP_ENCTYPE_NONE) {
198 return FALSE;
199 }
200
201 // Each uncaught Pokemon has an encounter
202 // chance weight based on its species.
203 // Compute the total weight for the current
204 // area. If it's zero, bail.
205 for (i = 0; i < CATCHING_SHOW_MONS; i++) {
206 if (catchingShow->caughtMonsOrder[i] == 0 && catchingShow->pokemon[i].area == area) {
207 totalRarity += catchingShow->pokemon[i].rarity;
208 }
209 }
210
211 if (totalRarity == 0) {
212 return FALSE;
213 }
214
215 // Prepend an extra bucket of weight 20
216 // for a no-encounter roll.
217 encounterChance = LCRandRange(totalRarity + WEIGHT_NO_ENCOUNTER);
218
219 if (encounterChance < WEIGHT_NO_ENCOUNTER) {
220 return FALSE;
221 }
222
223 encounterChance -= WEIGHT_NO_ENCOUNTER;
224
225 // Using the random number generated,
226 // select the bucket to serve as the
227 // encounter.
228 for (i = 0; i < CATCHING_SHOW_MONS; i++) {
229 if (catchingShow->caughtMonsOrder[i] == 0 && catchingShow->pokemon[i].area == area) {
230 if (encounterChance < catchingShow->pokemon[i].rarity) {
231 catchingShow->currentEncounterIndex = i;
232 return TRUE;
233 } else {
234 encounterChance -= catchingShow->pokemon[i].rarity;
235 }
236 }
237 }
238
239 // In theory, this is unreachable.
240 GF_ASSERT(FALSE);
241 return FALSE;
242}

278static u32 CalculateTypePoints(CatchingShow *catchingShow) {
279 int i, j;
280 int prevMonType1, prevMonType2, currMonType1, currMonType2;
281 u32 distinctTypeTracker = 0, totalTypePoints = 0;
282
283 // This score is calculated in two phases
284 // Phase 1: +200 points each time you don't
285 // catch two Pokemon sharing a type in a row.
286 // Maximum 1000 points.
287 for (i = 1; i < CATCHING_SHOW_MONS + 1; i++) {
288 for (j = 0; j < CATCHING_SHOW_MONS; j++) {
289 if (catchingShow->caughtMonsOrder[j] == i) {
290 currMonType1 = catchingShow->pokemon[j].type1;
291 currMonType2 = catchingShow->pokemon[j].type2;
292
293 if (i != 1
294 && prevMonType1 != currMonType1
295 && prevMonType1 != currMonType2
296 && prevMonType2 != currMonType1
297 && prevMonType2 != currMonType2) {
298 totalTypePoints += DIFFERENT_TYPE_BONUS;
299 }
300
301 prevMonType1 = currMonType1;
302 prevMonType2 = currMonType2;
303 distinctTypeTracker |= (1 << prevMonType1);
304 distinctTypeTracker |= (1 << prevMonType2);
305
306 break;
307 }
308 }
309 }
310
311 // Phase 2: +50 points for each unique
312 // type caught. Maximum 600 points.
313 while (distinctTypeTracker != 0) {
314 if (distinctTypeTracker & 1) {
315 totalTypePoints += DISTINCT_TYPE_BONUS;
316 }
317 distinctTypeTracker >>= 1;
318 }
319
320 return totalTypePoints;
321}
