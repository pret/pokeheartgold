165 ++(*state);

183 GameClearSave_PrintSaving(fieldSystem, env);
184 ++(*state);
185 break;

187 *state = 7;
188 }
189 break;
190 case 3:

192 ++(*state);
193 }
194 break;

197 int writeStatus = SaveGameNormal(fieldSystem->saveData);

203 ++(*state);
204 break;

244 Sound_SetMasterVolume(127);
245 Credits_LaunchApp(fieldSystem, &env->creditsArgs);
246 ++(*state);
247 }
248 break;
249 case 11:

251 Heap_Free(env);
252 Heap_Destroy(HEAP_ID_FIELD1);
253 OS_ResetSystem(0);

255 }

257 }
